   115→    global _auditor, _policy_generator, _edi_updater, _mttr_base_url

   121→    logging.basicConfig(
   122→        level=log_level,
   123→        format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
   124→    )
   125→
   126→    # Database
   127→    engine = create_engine_from_config(_config)
   128→    init_schema(engine)
   129→    _session_factory = get_session_factory(engine)
   130→
   131→    # Tools
   132→    _anonymiser = PIITokeniser(_config.anonymiser)

   135→    _policy_generator = PolicyGenerator(_config)

   138→    # MTTR tracker URL (Golang service)

   142→    )
   143→
   144→    # FastAPI app
   145→    app = FastAPI(
   146→        title="Maritime Global Compliance Swarm",
   147→        description=(
   148→            "Unified API gateway for maritime data governance and privacy compliance. "
   149→            "Exposes four compliance tools: PII Anonymiser, EDI SQL Auditor, "
   150→            "Remediation Route Generator, and MTTR Telemetry Tracker."
   151→        ),
   152→        version="1.0.0",
   153→        docs_url="/docs",
   154→        redoc_url="/redoc",
   155→    )
   156→
   157→    app.add_middleware(
   158→        CORSMiddleware,
   159→        allow_origins=["*"],
   160→        allow_credentials=True,
   161→        allow_methods=["*"],

   181→def _register_health_routes(app: FastAPI):
   182→    @app.get("/health", response_model=HealthResponse, tags=["health"])
   183→    async def health_check():
   184→        """Liveness probe — returns status of all four tools."""
   185→        tool_status = {"anonymiser": "ok", "auditor": "ok", "remediation": "ok"}
   186→
   187→        # Check MTTR tracker (Golang)
   188→        try:
   189→            async with httpx.AsyncClient(timeout=3.0) as client:
   190→                resp = await client.get(f"{_mttr_base_url}/health")
   191→                tool_status["mttr_tracker"] = "ok" if resp.status_code == 200 else "degraded"
   192→        except Exception:
   193→            tool_status["mttr_tracker"] = "unreachable"
   194→
   195→        return HealthResponse(
   196→            status="healthy",
   197→            service="compliance-swarm-gateway",
   198→            tools=tool_status,
   199→        )

   204→# ══════════════════════════════════════════════════════════════════════════

   206→def _register_anonymiser_routes(app: FastAPI):
   207→
   208→    @app.post(
   209→        "/api/v1/anonymise/manifest",
   210→        response_model=ManifestAnonymiseResponse,
   211→        responses={500: {"model": ErrorResponse}},
   212→        tags=["1. PII Anonymiser"],
   213→    )
   214→    async def anonymise_manifest(req: ManifestAnonymiseRequest):
   215→        """Tokenise all PII fields in a shipping manifest.
   216→
   217→        Accepts a raw manifest dict and replaces PII fields with
   218→        deterministic HMAC-SHA256 tokens. Anonymisation records are
   219→        persisted to the database for audit trail.
   220→        """
   221→        anonymised = _anonymiser.anonymise_manifest(req.manifest, req.manifest_id)
   222→
   223→        # Also anonymise any free-text fields
   224→        for ft_key in req.free_text_fields:
   225→            if ft_key in anonymised and isinstance(anonymised[ft_key], str):
   226→                anonymised[ft_key] = _anonymiser.anonymise_free_text(
   227→                    anonymised[ft_key], req.manifest_id
   228→                )
   229→
   230→        # Persist records to DB
   231→        records = _anonymiser.flush_records()
   232→        persisted = []
   233→        with _session_factory() as session:
   234→            for rec in records:
   235→                db_rec = AnonymisationRecord(
   236→                    manifest_id=rec["manifest_id"],
   237→                    field_name=rec["field_name"],
   238→                    field_category=rec["field_category"],
   239→                    original_hash=rec["original_hash"],
   240→                    token=rec["token"],
   241→                )
   242→                session.add(db_rec)
   243→                persisted.append(AnonymisedField(
   244→                    field_name=rec["field_name"],
   245→                    field_category=rec["field_category"].value,
   246→                    token=rec["token"],
   247→                    original_hash=rec["original_hash"],
   248→                ))
   249→
   250→        return ManifestAnonymiseResponse(
   251→            manifest_id=req.manifest_id,
   252→            anonymised_manifest=anonymised,
   253→            fields_processed=len(persisted),
   254→            records=persisted,
   255→        )
   256→
   257→    @app.post(
   258→        "/api/v1/anonymise/free-text",
   259→        response_model=FreeTextAnonymiseResponse,
   260→        responses={500: {"model": ErrorResponse}},
   261→        tags=["1. PII Anonymiser"],
   262→    )
   263→    async def anonymise_free_text(req: FreeTextAnonymiseRequest):
   264→        """Detect and tokenise PII patterns in free-text content.
   265→
   266→        Scans for email addresses, phone numbers, passport numbers,
   267→        and tax IDs, replacing matches with deterministic tokens.
   268→        """
   269→        result = _anonymiser.anonymise_free_text(req.text, req.manifest_id)
   270→        records = _anonymiser.flush_records()
   271→
   272→        persisted = []
   273→        with _session_factory() as session:
   274→            for rec in records:
   275→                db_rec = AnonymisationRecord(
   276→                    manifest_id=rec["manifest_id"],
   277→                    field_name=rec["field_name"],
   278→                    field_category=rec["field_category"],
   279→                    original_hash=rec["original_hash"],
   280→                    token=rec["token"],
   281→                )
   282→                session.add(db_rec)
   283→                persisted.append(AnonymisedField(
   284→                    field_name=rec["field_name"],
   285→                    field_category=rec["field_category"].value,
   286→                    token=rec["token"],
   287→                    original_hash=rec["original_hash"],
   288→                ))
   289→
   290→        return FreeTextAnonymiseResponse(
   291→            manifest_id=req.manifest_id,
   292→            anonymised_text=result,
   293→            patterns_found=len(persisted),
   294→            records=persisted,
   295→        )
   296→
   297→    @app.post(
   298→        "/api/v1/anonymise/scan",
   299→        response_model=ScanResponse,
   300→        responses={500: {"model": ErrorResponse}},
   301→        tags=["1. PII Anonymiser"],
   302→    )
   303→    async def scan_manifest(req: ScanRequest):
   304→        """Scan a manifest for PII fields without modifying data.
   305→
   306→        Returns all matching PII rules grouped by field name.
   307→        Supports jurisdiction filtering (GDPR, CCPA, LGPD, PDPA, PIPA).
   308→        """
   309→        jurisdiction = req.jurisdiction.value if req.jurisdiction else None
   310→        field_rules = _rule_engine.get_fields_for_manifest(req.manifest, jurisdiction)
   311→
   312→        matches: dict[str, list[ScanMatch]] = {}
   313→        for field_name, rules in field_rules.items():
   314→            matches[field_name] = [
   315→                ScanMatch(
   316→                    field_name=field_name,
   317→                    category=r.category,
   318→                    description=r.description,
   319→                    mandatory=r.mandatory,
   320→                    retention_max_days=r.retention_max_days,
   321→                    jurisdictions=[j.value for j in r.jurisdictions],
   322→                )
   323→                for r in rules
   324→            ]
   325→
   326→        return ScanResponse(
   327→            total_fields=len(req.manifest),
   328→            pii_fields_found=len(matches),
   329→            matches=matches,
   330→        )

   523→def _register_mttr_routes(app: FastAPI):
   524→
   525→    @app.post(
   526→        "/api/v1/mttr/events",
   527→        responses={
   528→            201: {"description": "Event recorded"},
   529→            400: {"model": ErrorResponse},
   530→            502: {"model": ErrorResponse},
   531→        },
   532→        tags=["4. MTTR Tracker"],
   533→    )
   534→    async def ingest_event(req: IngestEventRequest):
   535→        """Record a telemetry event for MTTR tracking.
   536→
   537→        Proxies to the Golang MTTR tracker service. Events track
   538→        the lifecycle of a finding through phases:
   539→        identified -> assigned -> in_progress -> resolved -> verified
   540→        """
   541→        try:
   542→            async with httpx.AsyncClient(timeout=10.0) as client:
   543→                resp = await client.post(
   544→                    f"{_mttr_base_url}/api/v1/events",
   545→                    json=req.model_dump(),
   546→                )
   547→                if resp.status_code != 201:
   548→                    raise HTTPException(
   549→                        status_code=resp.status_code,
   550→                        detail=resp.text,
   551→                    )
   552→                return resp.json()
   553→        except httpx.ConnectError:
   554→            raise HTTPException(
   555→                status_code=502,
   556→                detail=f"MTTR tracker unavailable at {_mttr_base_url}",
   557→            )
   558→
   559→    @app.get(
   560→        "/api/v1/mttr/findings/{finding_id}",
   561→        response_model=FindingMTTRResponse,
   562→        responses={404: {"model": ErrorResponse}, 502: {"model": ErrorResponse}},
   563→        tags=["4. MTTR Tracker"],
   564→    )
   565→    async def get_finding_mttr(finding_id: str):
   566→        """Get MTTR timeline and metrics for a specific finding.
   567→
   568→        Returns all lifecycle events and the computed MTTR in hours.
   569→        """
   570→        try:
   571→            async with httpx.AsyncClient(timeout=10.0) as client:
   572→                resp = await client.get(
   573→                    f"{_mttr_base_url}/api/v1/findings/{finding_id}"
   574→                )
   575→                if resp.status_code != 200:
   576→                    raise HTTPException(
   577→                        status_code=resp.status_code,
   578→                        detail=resp.text,
   579→                    )
   580→                data = resp.json()
   581→                return FindingMTTRResponse(
   582→                    finding_id=data["finding_id"],
   583→                    mttr_hours=data.get("mttr_hours"),
   584→                    event_count=data.get("event_count", 0),
   585→                    events=[
   586→                        MTTRTimelineEvent(
   587→                            id=e.get("id", ""),
   588→                            finding_id=e.get("finding_id", ""),
   589→                            phase=e.get("phase", ""),
   590→                            timestamp=e.get("timestamp", ""),
   591→                            assignee=e.get("assignee"),
   592→                            duration_seconds=e.get("duration_seconds"),
   593→                        )
   594→                        for e in data.get("events", [])
   595→                    ],
   596→                )
   597→        except httpx.ConnectError:
   598→            raise HTTPException(
   599→                status_code=502,
   600→                detail=f"MTTR tracker unavailable at {_mttr_base_url}",
   601→            )

   610→        from_date: Optional[str] = Query(None, description="ISO 8601 start (e.g. 2025-01-01T00:00:00Z)"),
   611→        to_date: Optional[str] = Query(None, description="ISO 8601 end"),

   623→            if to_date:

   632→                )
   633→                if resp.status_code != 200:
   634→                    raise HTTPException(
   635→                        status_code=resp.status_code,
   636→                        detail=resp.text,

   707→    async def list_findings(
   708→        severity: Optional[str] = Query(None),
   709→        status: Optional[str] = Query(None),
   710→        risk_category: Optional[str] = Query(None),
   711→        limit: int = Query(50, ge=1, le=500),
   712→        offset: int = Query(0, ge=0),
   713→    ):
   714→        """List audit findings with optional filtering.
   715→
   716→        Query all findings from the compliance database, filtered by
   717→        severity, status, or risk category. Supports pagination.
   718→        """
   719→        with _session_factory() as session:
   720→            query = session.query(AuditFinding)
   721→            if severity:
   722→                try:
   723→                    sev_enum = AuditSeverity(severity)
   724→                    query = query.filter(AuditFinding.severity == sev_enum)
   725→                except ValueError:
   726→                    pass
   727→            if status:
   728→                try:
   729→                    stat_enum = AuditStatus(status)
   730→                    query = query.filter(AuditFinding.status == stat_enum)
   731→                except ValueError:
   732→                    pass
   733→            if risk_category:
   734→                try:
   735→                    risk_enum = RiskCategory(risk_category)
   736→                    query = query.filter(AuditFinding.risk_category == risk_enum)
   737→                except ValueError:
   738→                    pass
   739→
   740→            total = query.count()
   741→            findings = query.order_by(AuditFinding.detected_at.desc()).offset(offset).limit(limit).all()
   742→
   743→            return FindingsListResponse(
   744→                total=total,
   745→                findings=[
   746→                    {
   747→                        "id": f.id,
   748→                        "finding_ref": f.finding_ref,
   749→                        "severity": f.severity.value if f.severity else None,
   750→                        "status": f.status.value if f.status else None,
   751→                        "risk_category": f.risk_category.value if f.risk_category else None,
   752→                        "title": f.title,
   753→                        "description": f.description,
   754→                        "affected_system": f.affected_system,
   755→                        "affected_table": f.affected_table,
   756→                        "affected_row_count": f.affected_row_count,
   757→                        "edi_standard": f.edi_standard.value if f.edi_standard else None,
   758→                        "detected_at": f.detected_at.isoformat() if f.detected_at else None,
   759→                        "remediated_at": f.remediated_at.isoformat() if f.remediated_at else None,
   760→                    }
   761→                    for f in findings
   762→                ],
   763→            )
