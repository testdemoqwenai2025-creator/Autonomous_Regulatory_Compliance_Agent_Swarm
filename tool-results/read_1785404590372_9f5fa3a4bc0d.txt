   168→
   169→function WaterfallChart({ rows, totalMs }: { rows: WaterfallRow[]; totalMs: number }) {
   170→  const scale = totalMs > 0 ? 100 / totalMs : 1;
   171→  return (
   172→    <div className="space-y-1.5">
   173→      {/* Time axis */}
   174→      <div className="flex text-[9px] text-slate-400 font-mono pl-[120px] pr-2">
   175→        <span>0ms</span>
   176→        <span className="ml-auto">{totalMs}ms</span>
   177→      </div>
   178→      {rows.map((r, i) => (
   179→        <div key={i} className="flex items-center gap-2">
   180→          <span className="text-[10px] font-medium text-slate-600 w-[115px] text-right flex-shrink-0 truncate" title={r.label}>{r.label}</span>
   181→          <div className="flex-1 relative h-5 bg-slate-100 rounded-sm overflow-hidden">
   182→            <div
   183→              className={`absolute top-0 h-full rounded-sm ${r.color} transition-all duration-500`}
   184→              style={{ left: `${Math.min(r.startMs * scale, 100)}%`, width: `${Math.max(r.widthMs * scale, 0.5)}%` }}
   185→              title={`${r.label}: ${r.widthMs}ms (starts at ${r.startMs}ms)`}
   186→            />
   187→          </div>
   188→          <span className="text-[10px] font-mono text-slate-500 w-[48px] text-right flex-shrink-0">{r.widthMs}ms</span>
   189→        </div>
   190→      ))}
   191→    </div>
   192→  );
   193→}

   209→  // Pipeline / trace state
   210→  const [pingData, setPingData] = useState<PingData | null>(null);
   211→  const [pingLoading, setPingLoading] = useState(false);
   212→  const [pingError, setPingError] = useState<string | null>(null);
   213→  const [postPayload, setPostPayload] = useState('');
   214→  const [postResult, setPostResult] = useState<string | null>(null);
   215→  const [responseHeaders, setResponseHeaders] = useState<Record<string, string>>({});
   216→  const [traceHistory, setTraceHistory] = useState<StoredTrace[]>([]);
   217→  const [loadingHistory, setLoadingHistory] = useState(false);
   218→  const renderStartRef = useRef<number>(0);
   219→
   220→  // ── Frontend Trace state (new) ──
   221→  const {
   222→    observedResources, stats: tracerStats,
   223→    traceAndPersist, getResourceTiming,
   224→  } = useRequestTracer();
   225→  const [correlatedResult, setCorrelatedResult] = useState<CorrelatedTraceResult | null>(null);

   228→  const [traceSummary, setTraceSummary] = useState<Record<string, unknown> | null>(null);
   229→  const [traceSummaryLoading, setTraceSummaryLoading] = useState(false);
   230→  const [fullTraceHistory, setFullTraceHistory] = useState<StoredTrace[]>([]);

   233→  const fetchData = useCallback(async () => {

   247→  }, []);
   248→
   249→  useEffect(() => { fetchData(); }, [fetchData]);
   250→
   251→  const loadTraceHistory = useCallback(async () => {
   252→    setLoadingHistory(true);
   253→    try {
   254→      const res = await fetch('/api/system/ping?trace=history');
   255→      const data = await res.json();
   256→      setTraceHistory(data.traces ?? []);
   257→    } catch (e) { console.error(e); }
   258→    finally { setLoadingHistory(false); }
   259→  }, []);
   260→
   261→  const runAudit = async () => {
   262→    setAuditing(true);
   263→    try {
   264→      const res = await fetch('/api/compliance/audit', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({}) });
   265→      const data = await res.json();
   266→      if (data.status === 'audit_complete') await fetchData();
   267→    } catch (e) { console.error(e); }
   268→    finally { setAuditing(false); }
   269→  };
   270→
   271→  // ── Instrumented ping with browser-side Performance API ──
   272→  const runPing = useCallback(async () => {
   273→    setPingLoading(true);
   274→    setPingError(null);
   275→    setPostResult(null);
   276→    renderStartRef.current = 0;
   277→
   278→    const fetchStart = performance.now();
   279→    const fetchStartIso = new Date().toISOString();
   280→
   281→    // Connection info (only available in some browsers)
   282→    const nav = navigator as Record<string, unknown>;
   283→    const connection = (nav.connection as Record<string, unknown>) ?? {};
   284→    const connectionType = (connection.effectiveType as string) ?? 'unknown';
   285→
   286→    // Try to get navigation type
   287→    let navigationType = 'unknown';
   288→    try {
   289→      const entries = performance.getEntriesByType?.('navigation') as Array<{ type: string }> | undefined;
   290→      if (entries?.[0]) navigationType = entries[0].type;
   291→    } catch { /* not available */ }
   292→
   293→    try {
   294→      const res = await fetch('/api/system/ping');
   295→      const ttfbMs = Math.round(performance.now() - fetchStart);
   296→
   297→      const jsonStart = performance.now();
   298→      const data: PingData = await res.json();
   299→      const jsonParseMs = Math.round(performance.now() - jsonStart);
   300→
   301→      const renderMark = performance.now();
   302→      const renderStartMs = Math.round(renderMark - fetchStart);
   303→      const roundTripMs = renderStartMs;
   304→
   305→      // Capture response headers
   306→      const headers: Record<string, string> = {};
   307→      res.headers.forEach((v, k) => { headers[k] = v; });
   308→      setResponseHeaders(headers);
   309→
   310→      // Now schedule state update timing measurement
   311→      requestAnimationFrame(() => {
   312→        const afterRender = Math.round(performance.now() - fetchStart);
   313→        renderStartRef.current = afterRender;
   314→      });
   315→
   316→      // Attach client timing to data so the waterfall can use it immediately
   317→      const clientTimingData = {
   318→        fetchStartIso,
   319→        ttfbMs,
   320→        roundTripMs,
   321→        jsonParseMs,
   322→        renderStartMs,
   323→        networkProtocol: '',
   324→        navigationType,
   325→        connectionType,
   326→      };
   327→
   328→      // Merge into pingData for immediate waterfall display
   329→      (data as Record<string, unknown>)._clientTimingRaw = clientTimingData;
   330→      setPingData(data);
   331→
   332→      // ── NOW re-send the client timing to a secondary endpoint for persistence ──
   333→      // We fire-and-forget a POST that carries the browser timings as a header
   334→      // so the server can persist a CorrelatedTrace.
   335→      // (The GET already ran, so we POST the timing separately.)
   336→      fetch('/api/system/ping', {
   337→        method: 'POST',
   338→        headers: {
   339→          'Content-Type': 'application/json',
   340→          'x-client-timing': JSON.stringify(clientTimingData),
   341→        },
   342→        body: JSON.stringify({ _timingOnly: true, _originalRequestId: data.server?.middleware?.requestId }),
   343→      }).then(r => r.json()).then(d => {
   344→        setPostResult(JSON.stringify(d, null, 2));
   345→      }).catch(() => {});
   346→
   347→      // Refresh trace history
   348→      setTimeout(() => loadTraceHistory(), 300);
   349→
   350→    } catch (e) {
   351→      setPingError(e instanceof Error ? e.message : 'Request failed');
   352→    } finally {
   353→      setPingLoading(false);
   354→    }
   355→  }, [loadTraceHistory]);
   356→
   357→  // ── POST with timing ──
   358→  const runPostPing = useCallback(async () => {
   359→    setPingLoading(true);
   360→    setPingError(null);
   361→    setPostResult(null);
   362→    const fetchStart = performance.now();
   363→    const fetchStartIso = new Date().toISOString();
   364→    try {
   365→      const payload = postPayload.trim() ? JSON.parse(postPayload) : { test: true, timestamp: new Date().toISOString() };
   366→      const res = await fetch('/api/system/ping', {
   367→        method: 'POST',
   368→        headers: {
   369→          'Content-Type': 'application/json',
   370→          'x-client-timing': JSON.stringify({
   371→            fetchStartIso,
   372→            ttfbMs: 0,
   373→            roundTripMs: Math.round(performance.now() - fetchStart),
   374→            jsonParseMs: 0,
   375→            renderStartMs: Math.round(performance.now() - fetchStart),
   376→            networkProtocol: '', navigationType: 'fetch', connectionType: 'unknown',
   377→          }),
   378→        },
   379→        body: JSON.stringify(payload),
   380→      });
   381→      const data = await res.json();
   382→      setPostResult(JSON.stringify(data, null, 2));
   383→      setTimeout(() => loadTraceHistory(), 300);
   384→    } catch (e) {
   385→      setPingError(e instanceof Error ? e.message : 'POST failed');
   386→    } finally {
   387→      setPingLoading(false);
   388→    }
   389→  }, [postPayload, loadTraceHistory]);
   390→
   391→  // ── CORRELATED TRACE: trace+persist using the useRequestTracer hook ──
   392→  const runCorrelatedTrace = useCallback(async () => {

   394→    setCorrelatedError(null);

   396→      const result = await traceAndPersist<CorrelatedTraceResult>(

   399→          method: 'POST',
   400→          headers: { 'Content-Type': 'application/json' },
   401→          body: JSON.stringify({ source: 'frontend_trace_tab', component: 'CorrelatedTracePanel' }),
   402→          source: 'frontend_trace_tab',
   403→          component: 'CorrelatedTracePanel',
   404→        },
   405→      );
   406→      setCorrelatedResult(result.data);
   407→    } catch (e) {
   408→      setCorrelatedError(e instanceof Error ? e.message : 'Correlated trace failed');
   409→    } finally {
   410→      setCorrelatedLoading(false);
   411→    }
   412→  }, [traceAndPersist]);
   413→
   414→  // ── Batch trace: trace ALL compliance endpoints at once ──
   415→  const [batchResults, setBatchResults] = useState<Array<{ url: string; ttfbMs: number; roundTripMs: number; protocol: string }>>([]);
   416→  const [batchLoading, setBatchLoading] = useState(false);
   417→
   418→  const runBatchTrace = useCallback(async () => {
   419→    setBatchLoading(true);
   420→    try {
   421→      const endpoints = [
   422→        '/api/compliance/health',
   423→        '/api/compliance/findings',
   424→        '/api/compliance/mttr',
   425→        '/api/compliance/policies',
   426→        '/api/compliance/profiles',
   427→      ];
   428→      const start = performance.now();
   429→      const results = await Promise.all(
   430→        endpoints.map(async (url) => {
   431→          const fetchStart = performance.now();
   432→          const fetchStartIso = new Date().toISOString();
   433→          try {
   434→            performance.clearResourceTimings(url);
   435→          } catch { /* ignore */ }
   436→          const res = await fetch(url);
   437→          const ttfbMs = Math.round(performance.now() - fetchStart);
   438→          const _ = await res.json(); // consume body
   439→          const roundTripMs = Math.round(performance.now() - fetchStart);
   440→          const rt = getResourceTiming(url);
   441→          return { url, ttfbMs, roundTripMs, protocol: rt?.nextHopProtocol ?? '' };
   442→        }),
   443→      );
   444→      const batchTotal = Math.round(performance.now() - start);
   445→      setBatchResults([...results, { url: `TOTAL (${endpoints.length} parallel)`, ttfbMs: batchTotal, roundTripMs: batchTotal, protocol: '' }]);
   446→    } catch (e) {
   447→      console.error('Batch trace error', e);
   448→    } finally {
   449→      setBatchLoading(false);
   450→    }
   451→  }, [getResourceTiming]);
   452→
   453→  const loadTraceSummary = useCallback(async () => {
   454→    setTraceSummaryLoading(true);
   455→    try {
   456→      const res = await fetch('/api/system/correlated-trace?mode=summary');
   457→      setTraceSummary(await res.json());
   458→    } catch (e) { console.error(e); }
   459→    finally { setTraceSummaryLoading(false); }
   460→  }, []);
   461→
   462→  const loadFullTraceHistory = useCallback(async () => {
   463→    setFullHistoryLoading(true);
   464→    try {
   465→      const res = await fetch('/api/system/correlated-trace?mode=history');
   466→      const data = await res.json();
   467→      setFullTraceHistory(data.traces ?? []);
   468→    } catch (e) { console.error(e); }
   469→    finally { setFullHistoryLoading(false); }
   470→  }, []);
   471→
   472→  const overall = mttr?.report.overall;
   473→  const openCritical = findings.filter(f => f.severity === 'critical' && (f.status === 'open' || f.status === 'in_progress')).length;
   474→
   475→  // ── Build waterfall rows from pingData ──
   476→  const buildWaterfall = (data: PingData): WaterfallRow[] => {
   477→    const clientRaw = (data as Record<string, unknown>)._clientTimingRaw as Record<string, number> | undefined;
   478→    const client = data.client;
   479→    const srv = data.server;
   480→    const corr = data.correlation;
   481→
   482→    const rows: WaterfallRow[] = [];
