   171→
   172→function WaterfallChart({ rows, totalMs }: { rows: WaterfallRow[]; totalMs: number }) {
   173→  const scale = totalMs > 0 ? 100 / totalMs : 1;
   174→  return (
   175→    <div className="space-y-1.5">
   176→      {/* Time axis */}
   177→      <div className="flex text-[9px] text-slate-400 font-mono pl-[120px] pr-2">
   178→        <span>0ms</span>
   179→        <span className="ml-auto">{totalMs}ms</span>
   180→      </div>
   181→      {rows.map((r, i) => (
   182→        <div key={i} className="flex items-center gap-2">
   183→          <span className="text-[10px] font-medium text-slate-600 w-[115px] text-right flex-shrink-0 truncate" title={r.label}>{r.label}</span>
   184→          <div className="flex-1 relative h-5 bg-slate-100 rounded-sm overflow-hidden">
   185→            <div
   186→              className={`absolute top-0 h-full rounded-sm ${r.color} transition-all duration-500`}
   187→              style={{ left: `${Math.min(r.startMs * scale, 100)}%`, width: `${Math.max(r.widthMs * scale, 0.5)}%` }}
   188→              title={`${r.label}: ${r.widthMs}ms (starts at ${r.startMs}ms)`}
   189→            />
   190→          </div>
   191→          <span className="text-[10px] font-mono text-slate-500 w-[48px] text-right flex-shrink-0">{r.widthMs}ms</span>
   192→        </div>
   193→      ))}
   194→    </div>
   195→  );
   196→}

   204→export default function ComplianceDashboard() {
   205→  const [findings, setFindings] = useState<Finding[]>([]);
   206→  const [mttr, setMttr] = useState<MTTRData | null>(null);
   207→  const [policies, setPolicies] = useState<Policy[]>([]);
   208→  const [profiles, setProfiles] = useState<Profile[]>([]);
   209→  const [loading, setLoading] = useState(true);
   210→  const [auditing, setAuditing] = useState(false);
   211→
   212→  // Pipeline / trace state
   213→  const [pingData, setPingData] = useState<PingData | null>(null);
   214→  const [pingLoading, setPingLoading] = useState(false);
   215→  const [pingError, setPingError] = useState<string | null>(null);
   216→  const [postPayload, setPostPayload] = useState('');
   217→  const [postResult, setPostResult] = useState<string | null>(null);
   218→  const [responseHeaders, setResponseHeaders] = useState<Record<string, string>>({});
   219→  const [traceHistory, setTraceHistory] = useState<StoredTrace[]>([]);
   220→  const [loadingHistory, setLoadingHistory] = useState(false);
   221→  const renderStartRef = useRef<number>(0);
   222→
   223→  // ── Frontend Trace state (new) ──
   224→  const {
   225→    observedResources, stats: tracerStats,
   226→    traceAndPersist, getResourceTiming,
   227→  } = useRequestTracer();
   228→  const [correlatedResult, setCorrelatedResult] = useState<CorrelatedTraceResult | null>(null);
   229→  const [correlatedLoading, setCorrelatedLoading] = useState(false);
   230→  const [correlatedError, setCorrelatedError] = useState<string | null>(null);
   231→  const [traceSummary, setTraceSummary] = useState<Record<string, unknown> | null>(null);
   232→  const [traceSummaryLoading, setTraceSummaryLoading] = useState(false);
   233→  const [fullTraceHistory, setFullTraceHistory] = useState<StoredTrace[]>([]);
   234→  const [fullHistoryLoading, setFullHistoryLoading] = useState(false);
   235→
   236→  // ── Full-Site Observability (ep-trace) ──
   237→  const {
   238→    runFullTrace, isTracing: epTracing,
   239→    lastTraceResult: epResult,
   240→    traceHistory: epHistory,
   241→    traceSummary: epSummary,
   242→    loadTraceHistory: loadEpHistory,
   243→    loadTraceSummary: loadEpSummary,
   244→  } = useEndpointTracer();
   245→  const [epHistoryLoaded, setEpHistoryLoaded] = useState(false);
   246→
   247→  const fetchData = useCallback(async () => {
   248→    setLoading(true);
   249→    try {
   250→      const [fRes, mRes, pRes, prRes] = await Promise.all([
   251→        fetch('/api/compliance/findings'), fetch('/api/compliance/mttr'),
   252→        fetch('/api/compliance/policies'), fetch('/api/compliance/profiles'),
   253→      ]);
   254→      const [fData, mData, pData, prData] = await Promise.all([
   255→        fRes.json(), mRes.json(), pRes.json(), prRes.json(),
   256→      ]);
   257→      setFindings(fData.findings); setMttr(mData);
   258→      setPolicies(pData.policies); setProfiles(prData.profiles);
   259→    } catch (e) { console.error(e); }
   260→    finally { setLoading(false); }
   261→  }, []);
   262→
   263→  useEffect(() => { fetchData(); }, [fetchData]);
   264→
   265→  const loadTraceHistory = useCallback(async () => {
   266→    setLoadingHistory(true);
   267→    try {
   268→      const res = await fetch('/api/system/ping?trace=history');
   269→      const data = await res.json();
   270→      setTraceHistory(data.traces ?? []);
   271→    } catch (e) { console.error(e); }
   272→    finally { setLoadingHistory(false); }
   273→  }, []);
   274→
   275→  const runAudit = async () => {
   276→    setAuditing(true);
   277→    try {
   278→      const res = await fetch('/api/compliance/audit', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({}) });
   279→      const data = await res.json();
   280→      if (data.status === 'audit_complete') await fetchData();
   281→    } catch (e) { console.error(e); }
   282→    finally { setAuditing(false); }
   283→  };
   284→
   285→  // ── Instrumented ping with browser-side Performance API ──
   286→  const runPing = useCallback(async () => {
   287→    setPingLoading(true);
   288→    setPingError(null);
   289→    setPostResult(null);
   290→    renderStartRef.current = 0;
   291→
   292→    const fetchStart = performance.now();
   293→    const fetchStartIso = new Date().toISOString();
   294→
   295→    // Connection info (only available in some browsers)
   296→    const nav = navigator as Record<string, unknown>;
   297→    const connection = (nav.connection as Record<string, unknown>) ?? {};
   298→    const connectionType = (connection.effectiveType as string) ?? 'unknown';
   299→
   300→    // Try to get navigation type
   301→    let navigationType = 'unknown';
   302→    try {
   303→      const entries = performance.getEntriesByType?.('navigation') as Array<{ type: string }> | undefined;
   304→      if (entries?.[0]) navigationType = entries[0].type;
   305→    } catch { /* not available */ }
   306→
   307→    try {
   308→      const res = await fetch('/api/system/ping');
   309→      const ttfbMs = Math.round(performance.now() - fetchStart);
   310→
   311→      const jsonStart = performance.now();
   312→      const data: PingData = await res.json();
   313→      const jsonParseMs = Math.round(performance.now() - jsonStart);
   314→
   315→      const renderMark = performance.now();
   316→      const renderStartMs = Math.round(renderMark - fetchStart);
   317→      const roundTripMs = renderStartMs;
   318→
   319→      // Capture response headers
   320→      const headers: Record<string, string> = {};
   321→      res.headers.forEach((v, k) => { headers[k] = v; });
   322→      setResponseHeaders(headers);
   323→
   324→      // Now schedule state update timing measurement
   325→      requestAnimationFrame(() => {
   326→        const afterRender = Math.round(performance.now() - fetchStart);
   327→        renderStartRef.current = afterRender;
   328→      });
   329→
   330→      // Attach client timing to data so the waterfall can use it immediately
   331→      const clientTimingData = {
   332→        fetchStartIso,
   333→        ttfbMs,
   334→        roundTripMs,
   335→        jsonParseMs,
   336→        renderStartMs,
   337→        networkProtocol: '',
   338→        navigationType,
   339→        connectionType,
   340→      };
   341→
   342→      // Merge into pingData for immediate waterfall display
   343→      (data as Record<string, unknown>)._clientTimingRaw = clientTimingData;
   344→      setPingData(data);
   345→
   346→      // ── NOW re-send the client timing to a secondary endpoint for persistence ──
   347→      // We fire-and-forget a POST that carries the browser timings as a header
   348→      // so the server can persist a CorrelatedTrace.
   349→      // (The GET already ran, so we POST the timing separately.)
   350→      fetch('/api/system/ping', {
   351→        method: 'POST',
   352→        headers: {
   353→          'Content-Type': 'application/json',
   354→          'x-client-timing': JSON.stringify(clientTimingData),
   355→        },
   356→        body: JSON.stringify({ _timingOnly: true, _originalRequestId: data.server?.middleware?.requestId }),
   357→      }).then(r => r.json()).then(d => {
   358→        setPostResult(JSON.stringify(d, null, 2));
   359→      }).catch(() => {});
   360→
   361→      // Refresh trace history
   362→      setTimeout(() => loadTraceHistory(), 300);
   363→
   364→    } catch (e) {
   365→      setPingError(e instanceof Error ? e.message : 'Request failed');
   366→    } finally {
   367→      setPingLoading(false);
   368→    }
   369→  }, [loadTraceHistory]);
   370→
   371→  // ── POST with timing ──
   372→  const runPostPing = useCallback(async () => {
   373→    setPingLoading(true);
   374→    setPingError(null);
   375→    setPostResult(null);
   376→    const fetchStart = performance.now();
   377→    const fetchStartIso = new Date().toISOString();
   378→    try {
   379→      const payload = postPayload.trim() ? JSON.parse(postPayload) : { test: true, timestamp: new Date().toISOString() };
   380→      const res = await fetch('/api/system/ping', {
   381→        method: 'POST',
   382→        headers: {
   383→          'Content-Type': 'application/json',
   384→          'x-client-timing': JSON.stringify({
   385→            fetchStartIso,
   386→            ttfbMs: 0,
   387→            roundTripMs: Math.round(performance.now() - fetchStart),
   388→            jsonParseMs: 0,
   389→            renderStartMs: Math.round(performance.now() - fetchStart),
   390→            networkProtocol: '', navigationType: 'fetch', connectionType: 'unknown',
   391→          }),
   392→        },
   393→        body: JSON.stringify(payload),
   394→      });
   395→      const data = await res.json();
   396→      setPostResult(JSON.stringify(data, null, 2));
   397→      setTimeout(() => loadTraceHistory(), 300);
   398→    } catch (e) {
   399→      setPingError(e instanceof Error ? e.message : 'POST failed');
   400→    } finally {
   401→      setPingLoading(false);
   402→    }
   403→  }, [postPayload, loadTraceHistory]);
   404→
   405→  // ── CORRELATED TRACE: trace+persist using the useRequestTracer hook ──
   406→  const runCorrelatedTrace = useCallback(async () => {
   407→    setCorrelatedLoading(true);
   408→    setCorrelatedError(null);
   409→    try {
   410→      const result = await traceAndPersist<CorrelatedTraceResult>(
   411→        '/api/system/correlated-trace',
   412→        {
   413→          method: 'POST',
   414→          headers: { 'Content-Type': 'application/json' },
   415→          body: JSON.stringify({ source: 'frontend_trace_tab', component: 'CorrelatedTracePanel' }),
   416→          source: 'frontend_trace_tab',
   417→          component: 'CorrelatedTracePanel',
   418→        },
   419→      );
   420→      setCorrelatedResult(result.data);
   421→    } catch (e) {
   422→      setCorrelatedError(e instanceof Error ? e.message : 'Correlated trace failed');
   423→    } finally {
   424→      setCorrelatedLoading(false);
   425→    }
   426→  }, [traceAndPersist]);
   427→
   428→  // ── Batch trace: trace ALL compliance endpoints at once ──
   429→  const [batchResults, setBatchResults] = useState<Array<{ url: string; ttfbMs: number; roundTripMs: number; protocol: string }>>([]);
   430→  const [batchLoading, setBatchLoading] = useState(false);
   431→
   432→  const runBatchTrace = useCallback(async () => {
   433→    setBatchLoading(true);
   434→    try {
   435→      const endpoints = [
   436→        '/api/compliance/health',
   437→        '/api/compliance/findings',
   438→        '/api/compliance/mttr',
   439→        '/api/compliance/policies',
   440→        '/api/compliance/profiles',
   441→      ];
   442→      const start = performance.now();
   443→      const results = await Promise.all(
   444→        endpoints.map(async (url) => {
   445→          const fetchStart = performance.now();
   446→          const fetchStartIso = new Date().toISOString();
   447→          try {
   448→            const perfAny = performance as unknown as { clearResourceTimings?: (name?: string) => void };
   449→            perfAny.clearResourceTimings?.(url);
   450→          } catch { /* ignore */ }
   451→          const res = await fetch(url);
   452→          const ttfbMs = Math.round(performance.now() - fetchStart);
   453→          const _ = await res.json(); // consume body
   454→          const roundTripMs = Math.round(performance.now() - fetchStart);
   455→          const rt = getResourceTiming(url);
   456→          return { url, ttfbMs, roundTripMs, protocol: rt?.nextHopProtocol ?? '' };
   457→        }),
   458→      );
   459→      const batchTotal = Math.round(performance.now() - start);
   460→      setBatchResults([...results, { url: `TOTAL (${endpoints.length} parallel)`, ttfbMs: batchTotal, roundTripMs: batchTotal, protocol: '' }]);
   461→    } catch (e) {
   462→      console.error('Batch trace error', e);
   463→    } finally {
   464→      setBatchLoading(false);
   465→    }
   466→  }, [getResourceTiming]);
   467→
   468→  const loadTraceSummary = useCallback(async () => {
   469→    setTraceSummaryLoading(true);
   470→    try {
   471→      const res = await fetch('/api/system/correlated-trace?mode=summary');
   472→      setTraceSummary(await res.json());
   473→    } catch (e) { console.error(e); }
   474→    finally { setTraceSummaryLoading(false); }
   475→  }, []);
   476→
   477→  const loadFullTraceHistory = useCallback(async () => {
   478→    setFullHistoryLoading(true);
   479→    try {
   480→      const res = await fetch('/api/system/correlated-trace?mode=history');
   481→      const data = await res.json();
