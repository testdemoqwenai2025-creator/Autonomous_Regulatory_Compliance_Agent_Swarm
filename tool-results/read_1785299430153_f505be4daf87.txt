   540→        records = _anonymiser.flush_records()

   547→                    field_category=rec["field_category"],
   548→                    original_hash=rec["original_hash"],
   549→                    token=rec["token"],
   550→                )
   551→                session.add(db_rec)
   552→                persisted.append(AnonymisedField(
   553→                    field_name=rec["field_name"],
   554→                    field_category=rec["field_category"].value,
