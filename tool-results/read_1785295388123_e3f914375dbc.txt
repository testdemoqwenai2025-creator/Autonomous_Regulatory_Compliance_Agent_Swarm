   245→def _register_connectivity_route(app: FastAPI):
   246→
   247→    @app.get(
   248→        "/api/v1/system/connectivity",
   249→        response_model=ConnectivityResponse,
   250→        tags=["system"],
   251→    )
   252→    async def connectivity_check():
   253→        """Comprehensive connectivity diagnostics for the frontend.
   254→
   255→        Tests every backend component and middleware connection, returning
   256→        detailed status with latency measurements. The frontend calls this
   257→        endpoint to confirm it can effectively communicate with all backend
   258→        services, tools, and the Golang MTTR tracker.
   259→
   260→        Returns per-component status: ok | degraded | unavailable.
   261→        """
   262→        from sqlalchemy import text as sa_text
   263→
   264→        ts = datetime.now(timezone.utc).isoformat()
   265→        components = {}
   266→        all_ok = True
   267→
   268→        # 1. Database connectivity
   269→        db_status = "ok"
   270→        db_latency = None
   271→        db_detail = None
   272→        try:
   273→            t0 = time.monotonic()
   274→            with _session_factory() as session:
   275→                session.execute(sa_text("SELECT 1")).scalar()
   276→            db_latency = round((time.monotonic() - t0) * 1000, 2)
   277→        except Exception as e:
   278→            db_status = "unavailable"
   279→            db_detail = str(e)[:200]
   280→            all_ok = False
   281→        database_status = ComponentStatus(
   282→            name="Compliance Database",
   283→            status=db_status,
   284→            latency_ms=db_latency,
   285→            detail=db_detail,
   286→        )
   287→        components["database"] = database_status
   288→
   289→        # 2. PII Anonymiser
   290→        t0 = time.monotonic()
   291→        try:
   292→            _anonymiser._vault.tokenise("connectivity-test", PIIFieldCategory.CONTACT_INFO)
   293→            anonymiser_status = "ok"
   294→        except Exception:
   295→            anonymiser_status = "unavailable"
   296→            all_ok = False
   297→        anonymiser_latency = round((time.monotonic() - t0) * 1000, 2)
   298→        components["anonymiser"] = ComponentStatus(
   299→            name="PII Anonymiser",
   300→            status=anonymiser_status,
   301→            latency_ms=anonymiser_latency,
   302→        )
   303→
   304→        # 3. NER Detector
   305→        t0 = time.monotonic()
   306→        try:
   307→            _ner_detector.detect("Test text")
   308→            ner_status = "ok"
   309→        except Exception:
   310→            ner_status = "unavailable"
   311→            all_ok = False
   312→        ner_latency = round((time.monotonic() - t0) * 1000, 2)
   313→        components["ner_detector"] = ComponentStatus(
   314→            name="NER Detector",
   315→            status=ner_status,
   316→            latency_ms=ner_latency,
   317→            detail=f"layers: {_ner_detector.layers_available}, spacy: {_ner_detector.spacy_available}",
   318→        )
   319→
   320→        # 4. EDI Auditor
   321→        t0 = time.monotonic()
   322→        try:
   323→            _rule_engine.find_matching_rules("consignee_name")
   324→            auditor_status = "ok"
   325→        except Exception:
   326→            auditor_status = "unavailable"
   327→            all_ok = False
   328→        auditor_latency = round((time.monotonic() - t0) * 1000, 2)
   329→        components["auditor"] = ComponentStatus(
   330→            name="EDI SQL Auditor",
   331→            status=auditor_status,
   332→            latency_ms=auditor_latency,
   333→        )
   334→
   335→        # 5. Remediation Generator
   336→        t0 = time.monotonic()
   337→        try:
   338→            from remediation.policy_gen import REMEDIATION_MATRIX
   339→            _ = len(REMEDIATION_MATRIX)
   340→            remediation_status = "ok"
   341→        except Exception:
   342→            remediation_status = "unavailable"
   343→            all_ok = False
   344→        remediation_latency = round((time.monotonic() - t0) * 1000, 2)
   345→        components["remediation"] = ComponentStatus(
   346→            name="Remediation Generator",
   347→            status=remediation_status,
   348→            latency_ms=remediation_latency,
   349→        )
   350→
   351→        # 6. MTTR Proxy (Golang)
   352→        mttr_status = "unavailable"
   353→        mttr_latency = None
   354→        mttr_detail = None
   355→        try:
   356→            t0 = time.monotonic()
   357→            async with httpx.AsyncClient(timeout=5.0) as client:
   358→                resp = await client.get(f"{_mttr_base_url}/health")
   359→                mttr_latency = round((time.monotonic() - t0) * 1000, 2)
   360→                if resp.status_code == 200:
   361→                    mttr_status = "ok"
   362→                else:
   363→                    mttr_status = "degraded"
   364→                    mttr_detail = f"HTTP {resp.status_code}"
   365→        except httpx.ConnectError:
   366→            mttr_detail = f"Cannot connect to {_mttr_base_url}"
   367→        except Exception as e:
   368→            mttr_detail = str(e)[:200]
   369→
   370→        if mttr_status == "unavailable":
   371→            all_ok = False
   372→
   373→        mttr_component = ComponentStatus(
   374→            name="MTTR Tracker (Go)",
   375→            status=mttr_status,
   376→            latency_ms=mttr_latency,
   377→            detail=mttr_detail,
   378→        )
   379→        components["mttr_tracker"] = mttr_component
   380→
   381→        # 7. Query Registry
   382→        t0 = time.monotonic()
   383→        try:
   384→            stats = _query_registry.get_statistics()
   385→            registry_status = "ok"
   386→            registry_detail = f"{stats['active_queries']} active, {stats['builtin_queries']} builtin"
   387→        except Exception as e:
   388→            registry_status = "unavailable"
   389→            registry_detail = str(e)[:200]
   390→            all_ok = False
   391→        registry_latency = round((time.monotonic() - t0) * 1000, 2)
   392→        registry_component = ComponentStatus(
   393→            name="Query Registry",
   394→            status=registry_status,
   395→            latency_ms=registry_latency,
   396→            detail=registry_detail,
   397→        )
   398→        components["query_registry"] = registry_component
   399→
   400→        overall = "ok" if all_ok else ("degraded" if db_status == "ok" else "down")
   401→
   402→        # Count routes
   403→        total_routes = len(app.routes)
   404→        active_routes = sum(
   405→            1 for r in app.routes
   406→            if hasattr(r, "methods") and hasattr(r, "path")
   407→        )
   408→
   409→        return ConnectivityResponse(
   410→            status=overall,
   411→            timestamp=ts,
   412→            gateway_version="1.1.0",
   413→            components=components,
   414→            database=database_status,
   415→            ner_layers=_ner_detector.layers_available,
   416→            mttr_proxy=mttr_component,
   417→            query_registry=registry_component,
   418→            active_routes=active_routes,
   419→            total_routes=total_routes,
   420→        )

   427→def _register_anonymiser_routes(app: FastAPI):
   428→
   429→    @app.post(
   430→        "/api/v1/anonymise/manifest",
   431→        response_model=ManifestAnonymiseResponse,
   432→        responses={500: {"model": ErrorResponse}},
   433→        tags=["1. PII Anonymiser"],
   434→    )
   435→    async def anonymise_manifest(req: ManifestAnonymiseRequest):
   436→        """Tokenise all PII fields in a shipping manifest.
   437→
   438→        Accepts a raw manifest dict and replaces PII fields with
   439→        deterministic HMAC-SHA256 tokens. Anonymisation records are
   440→        persisted to the database for audit trail.
   441→        """
   442→        anonymised = _anonymiser.anonymise_manifest(req.manifest, req.manifest_id)
   443→
   444→        # Also anonymise any free-text fields
   445→        for ft_key in req.free_text_fields:
   446→            if ft_key in anonymised and isinstance(anonymised[ft_key], str):
   447→                anonymised[ft_key] = _anonymiser.anonymise_free_text(
   448→                    anonymised[ft_key], req.manifest_id
   449→                )
   450→
   451→        # Persist records to DB
   452→        records = _anonymiser.flush_records()
   453→        persisted = []
   454→        with _session_factory() as session:
   455→            for rec in records:
   456→                db_rec = AnonymisationRecord(
   457→                    manifest_id=rec["manifest_id"],
   458→                    field_name=rec["field_name"],
   459→                    field_category=rec["field_category"],
   460→                    original_hash=rec["original_hash"],
   461→                    token=rec["token"],
   462→                )
   463→                session.add(db_rec)
   464→                persisted.append(AnonymisedField(
   465→                    field_name=rec["field_name"],
   466→                    field_category=rec["field_category"].value,
   467→                    token=rec["token"],
   468→                    original_hash=rec["original_hash"],
   469→                ))
   470→
   471→        return ManifestAnonymiseResponse(
   472→            manifest_id=req.manifest_id,
   473→            anonymised_manifest=anonymised,
   474→            fields_processed=len(persisted),
   475→            records=persisted,
   476→        )
   477→
   478→    @app.post(
   479→        "/api/v1/anonymise/free-text",
   480→        response_model=FreeTextAnonymiseResponse,
   481→        responses={500: {"model": ErrorResponse}},
   482→        tags=["1. PII Anonymiser"],
   483→    )
   484→    async def anonymise_free_text(req: FreeTextAnonymiseRequest):
   485→        """Detect and tokenise PII patterns in free-text content.
   486→
   487→        Scans for email addresses, phone numbers, passport numbers,
   488→        and tax IDs, replacing matches with deterministic tokens.
   489→        """
   490→        result = _anonymiser.anonymise_free_text(req.text, req.manifest_id)
   491→        records = _anonymiser.flush_records()
   492→
   493→        persisted = []
   494→        with _session_factory() as session:
   495→            for rec in records:
   496→                db_rec = AnonymisationRecord(
   497→                    manifest_id=rec["manifest_id"],
   498→                    field_name=rec["field_name"],
   499→                    field_category=rec["field_category"],
   500→                    original_hash=rec["original_hash"],
   501→                    token=rec["token"],
   502→                )
   503→                session.add(db_rec)
   504→                persisted.append(AnonymisedField(
   505→                    field_name=rec["field_name"],
   506→                    field_category=rec["field_category"].value,
   507→                    token=rec["token"],
   508→                    original_hash=rec["original_hash"],
   509→                ))
   510→
   511→        return FreeTextAnonymiseResponse(
   512→            manifest_id=req.manifest_id,
   513→            anonymised_text=result,
   514→            patterns_found=len(persisted),
   515→            records=persisted,
   516→        )
   517→
   518→    @app.post(
   519→        "/api/v1/anonymise/scan",
   520→        response_model=ScanResponse,
   521→        responses={500: {"model": ErrorResponse}},
   522→        tags=["1. PII Anonymiser"],
   523→    )
   524→    async def scan_manifest(req: ScanRequest):
   525→        """Scan a manifest for PII fields without modifying data.
   526→
   527→        Returns all matching PII rules grouped by field name.
   528→        Supports jurisdiction filtering (GDPR, CCPA, LGPD, PDPA, PIPA).
   529→        """
   530→        jurisdiction = req.jurisdiction.value if req.jurisdiction else None
   531→        field_rules = _rule_engine.get_fields_for_manifest(req.manifest, jurisdiction)
   532→
   533→        matches: dict[str, list[ScanMatch]] = {}
   534→        for field_name, rules in field_rules.items():
   535→            matches[field_name] = [
   536→                ScanMatch(
   537→                    field_name=field_name,
   538→                    category=r.category,
   539→                    description=r.description,
   540→                    mandatory=r.mandatory,
   541→                    retention_max_days=r.retention_max_days,
   542→                    jurisdictions=[j.value for j in r.jurisdictions],
   543→                )
   544→                for r in rules
   545→            ]
   546→
   547→        return ScanResponse(
   548→            total_fields=len(req.manifest),
   549→            pii_fields_found=len(matches),
   550→            matches=matches,
   551→        )

   558→def _register_auditor_routes(app: FastAPI):

   566→    async def run_audit(req: RunAuditRequest):
   567→        """Execute compliance audit queries against the FMS database.
   568→
   569→        Runs 11 parametric SQL queries across 5 compliance domains:
   570→        encryption, customs documentation, EDI format, data retention,
   571→        and access control. Results are persisted as AuditFindings.
   572→        """
   573→        domain = None
   574→        if req.domain:
   575→            domain = ComplianceDomain(req.domain.value)
   576→
   577→        min_sev = req.min_severity.value if req.min_severity else None
   578→
   579→        findings = _auditor.run_audit(domain=domain, min_severity=min_sev)
   580→
   581→        return RunAuditResponse(
   582→            queries_executed=len(ALL_AUDIT_QUERIES),
   583→            findings_count=len(findings),
   584→            findings=[
   585→                AuditFindingSummary(
   586→                    finding_ref=f.get("finding_ref", ""),
   587→                    severity=f.get("severity", ""),
   588→                    affected_row_count=f.get("affected_row_count", 0),
   589→                    title=f.get("title", ""),
   590→                    query_id=f.get("query_id"),
   591→                    error=f.get("error"),
   592→                )
   593→                for f in findings
   594→            ],
   595→        )
