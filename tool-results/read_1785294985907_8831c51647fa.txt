   219→def _register_anonymiser_routes(app: FastAPI):
   220→
   221→    @app.post(
   222→        "/api/v1/anonymise/manifest",
   223→        response_model=ManifestAnonymiseResponse,
   224→        responses={500: {"model": ErrorResponse}},
   225→        tags=["1. PII Anonymiser"],
   226→    )
   227→    async def anonymise_manifest(req: ManifestAnonymiseRequest):
   228→        """Tokenise all PII fields in a shipping manifest.
   229→
   230→        Accepts a raw manifest dict and replaces PII fields with
   231→        deterministic HMAC-SHA256 tokens. Anonymisation records are
   232→        persisted to the database for audit trail.
   233→        """
   234→        anonymised = _anonymiser.anonymise_manifest(req.manifest, req.manifest_id)
   235→
   236→        # Also anonymise any free-text fields
   237→        for ft_key in req.free_text_fields:
   238→            if ft_key in anonymised and isinstance(anonymised[ft_key], str):
   239→                anonymised[ft_key] = _anonymiser.anonymise_free_text(
   240→                    anonymised[ft_key], req.manifest_id
   241→                )
   242→
   243→        # Persist records to DB
   244→        records = _anonymiser.flush_records()
   245→        persisted = []
   246→        with _session_factory() as session:
   247→            for rec in records:
   248→                db_rec = AnonymisationRecord(
   249→                    manifest_id=rec["manifest_id"],
   250→                    field_name=rec["field_name"],
   251→                    field_category=rec["field_category"],
   252→                    original_hash=rec["original_hash"],
   253→                    token=rec["token"],
   254→                )
   255→                session.add(db_rec)
   256→                persisted.append(AnonymisedField(
   257→                    field_name=rec["field_name"],
   258→                    field_category=rec["field_category"].value,
   259→                    token=rec["token"],
   260→                    original_hash=rec["original_hash"],
   261→                ))
   262→
   263→        return ManifestAnonymiseResponse(
   264→            manifest_id=req.manifest_id,
   265→            anonymised_manifest=anonymised,
   266→            fields_processed=len(persisted),
   267→            records=persisted,
   268→        )
   269→
   270→    @app.post(
   271→        "/api/v1/anonymise/free-text",
   272→        response_model=FreeTextAnonymiseResponse,
   273→        responses={500: {"model": ErrorResponse}},
   274→        tags=["1. PII Anonymiser"],
   275→    )
   276→    async def anonymise_free_text(req: FreeTextAnonymiseRequest):
   277→        """Detect and tokenise PII patterns in free-text content.
   278→
   279→        Scans for email addresses, phone numbers, passport numbers,
   280→        and tax IDs, replacing matches with deterministic tokens.
   281→        """
   282→        result = _anonymiser.anonymise_free_text(req.text, req.manifest_id)
   283→        records = _anonymiser.flush_records()
   284→
   285→        persisted = []
   286→        with _session_factory() as session:
   287→            for rec in records:
   288→                db_rec = AnonymisationRecord(
   289→                    manifest_id=rec["manifest_id"],
   290→                    field_name=rec["field_name"],
   291→                    field_category=rec["field_category"],
   292→                    original_hash=rec["original_hash"],
   293→                    token=rec["token"],
   294→                )
   295→                session.add(db_rec)
   296→                persisted.append(AnonymisedField(
   297→                    field_name=rec["field_name"],
   298→                    field_category=rec["field_category"].value,
   299→                    token=rec["token"],
   300→                    original_hash=rec["original_hash"],
   301→                ))
   302→
   303→        return FreeTextAnonymiseResponse(
   304→            manifest_id=req.manifest_id,
   305→            anonymised_text=result,
   306→            patterns_found=len(persisted),
   307→            records=persisted,
   308→        )
   309→
   310→    @app.post(
   311→        "/api/v1/anonymise/scan",
   312→        response_model=ScanResponse,
   313→        responses={500: {"model": ErrorResponse}},
   314→        tags=["1. PII Anonymiser"],
   315→    )
   316→    async def scan_manifest(req: ScanRequest):
   317→        """Scan a manifest for PII fields without modifying data.
   318→
   319→        Returns all matching PII rules grouped by field name.
   320→        Supports jurisdiction filtering (GDPR, CCPA, LGPD, PDPA, PIPA).
   321→        """
   322→        jurisdiction = req.jurisdiction.value if req.jurisdiction else None
   323→        field_rules = _rule_engine.get_fields_for_manifest(req.manifest, jurisdiction)
   324→
   325→        matches: dict[str, list[ScanMatch]] = {}
   326→        for field_name, rules in field_rules.items():
   327→            matches[field_name] = [
   328→                ScanMatch(
   329→                    field_name=field_name,
   330→                    category=r.category,
   331→                    description=r.description,
   332→                    mandatory=r.mandatory,
   333→                    retention_max_days=r.retention_max_days,
   334→                    jurisdictions=[j.value for j in r.jurisdictions],
   335→                )
   336→                for r in rules
   337→            ]
   338→
   339→        return ScanResponse(
   340→            total_fields=len(req.manifest),
   341→            pii_fields_found=len(matches),
   342→            matches=matches,
   343→        )

   715→    @app.get(
   716→        "/api/v1/findings",
   717→        response_model=FindingsListResponse,
   718→        tags=["findings"],
   719→    )
   720→    async def list_findings(
   721→        severity: Optional[str] = Query(None),
   722→        status: Optional[str] = Query(None),
   723→        risk_category: Optional[str] = Query(None),
   724→        limit: int = Query(50, ge=1, le=500),
   725→        offset: int = Query(0, ge=0),
   726→    ):
   727→        """List audit findings with optional filtering.
   728→
   729→        Query all findings from the compliance database, filtered by
   730→        severity, status, or risk category. Supports pagination.
   731→        """
   732→        with _session_factory() as session:
   733→            query = session.query(AuditFinding)
   734→            if severity:
   735→                try:
   736→                    sev_enum = AuditSeverity(severity)
   737→                    query = query.filter(AuditFinding.severity == sev_enum)
   738→                except ValueError:
   739→                    pass
   740→            if status:
   741→                try:
   742→                    stat_enum = AuditStatus(status)
   743→                    query = query.filter(AuditFinding.status == stat_enum)
   744→                except ValueError:
   745→                    pass
   746→            if risk_category:
   747→                try:
   748→                    risk_enum = RiskCategory(risk_category)
   749→                    query = query.filter(AuditFinding.risk_category == risk_enum)
   750→                except ValueError:
   751→                    pass
   752→
   753→            total = query.count()
   754→            findings = query.order_by(AuditFinding.detected_at.desc()).offset(offset).limit(limit).all()
   755→
   756→            return FindingsListResponse(
   757→                total=total,
   758→                findings=[
   759→                    {
   760→                        "id": f.id,
   761→                        "finding_ref": f.finding_ref,
   762→                        "severity": f.severity.value if f.severity else None,
   763→                        "status": f.status.value if f.status else None,
   764→                        "risk_category": f.risk_category.value if f.risk_category else None,
   765→                        "title": f.title,
   766→                        "description": f.description,
   767→                        "affected_system": f.affected_system,
   768→                        "affected_table": f.affected_table,
   769→                        "affected_row_count": f.affected_row_count,
   770→                        "edi_standard": f.edi_standard.value if f.edi_standard else None,
   771→                        "detected_at": f.detected_at.isoformat() if f.detected_at else None,
   772→                        "remediated_at": f.remediated_at.isoformat() if f.remediated_at else None,
   773→                    }
   774→                    for f in findings
   775→                ],
   776→            )
