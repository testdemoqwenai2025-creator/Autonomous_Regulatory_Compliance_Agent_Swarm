   761→    for (const state of smDef.states) {
   762→      const transNames = state.transitions.map(t => t.to_state.replace(/_/g,' ')).join(', ') || '—';
   763→      const triggers = state.transitions.flatMap(t => t.triggers).join(', ') || '—';
   764→      const signoff = state.transitions.some(t => t.requires_sign_off) ? '⚠️ Yes' : '—';
   765→      const tos = state.timeouts_hours;
   766→      const timeoutStr = tos ? `${tos.critical || '—'}h / ${tos.high || '—'}h / ${tos.medium || '—'}h` : '—';
   767→      tableHtml += `<tr>
   768→        <td><span class="badge badge-${state.name}">${state.name.replace(/_/g,' ')}</span></td>
   769→        <td style="font-size:12px">${transNames}</td>
   770→        <td style="font-size:11px; color:var(--text-dim)">${triggers}</td>
   771→        <td>${signoff}</td>
   772→        <td style="font-family:monospace; font-size:12px">${timeoutStr}</td>
   773→      </tr>`;
   774→    }
   775→    tableHtml += '</table>';
   776→    document.getElementById('smStatesTable').innerHTML = tableHtml;
   777→
   778→    // Enable finding explorer
   779→    document.getElementById('btnTransition').disabled = false;
   780→    document.getElementById('btnTimeline').disabled = false;

   782→    toast(`State machine loaded: ${smDef.total_states} states, ${smDef.total_transitions} transitions`, 'success');

   784→    toast('Failed to load state machine: ' + e.message, 'error');
   785→  }

   813→async function doTransition() {
   814→  const findingId = document.getElementById('smFindingId').value.trim();
   815→  const targetState = document.getElementById('smTargetState').value;
   816→  const trigger = document.getElementById('smTrigger').value;
   817→  const actor = document.getElementById('smActor').value;
   818→  if (!findingId || !targetState || !trigger) { toast('Fill in finding ID, target state, and trigger', 'error'); return; }
   819→
   820→  try {
   821→    const r = await fetch(API + `/api/v1/state-machine/transitions/${findingId}`, {
   822→      method: 'POST',
   823→      headers: {'Content-Type': 'application/json'},
   824→      body: JSON.stringify({ target_state: targetState, trigger, actor, context: {} })
   825→    });
   826→    const d = await r.json();
   827→    const color = d.success ? 'var(--green)' : 'var(--red)';
   828→    document.getElementById('smTransitionResult').innerHTML = `
   829→      <div style="padding:14px; background:var(--navy-light); border:1px solid ${color}; border-radius:7px; margin-bottom:12px">
   830→        <div style="font-weight:600; color:${color}; margin-bottom:6px">${d.success ? 'Transition Successful' : 'Transition Failed'}</div>
   831→        <div style="font-size:12px; font-family:monospace">
   832→          ${d.from_state} → ${d.to_state} | trigger: ${d.trigger} | actor: ${d.actor}<br>
   833→          ${d.error ? 'Error: ' + d.error : ''}
   834→          ${d.guard_failed ? 'GUARD CONDITION FAILED' : ''}
   835→          ${d.auto_escalated ? 'AUTO-ESCALATED' : ''}
   836→          ${d.timeout_hours ? 'Timeout: ' + d.timeout_hours + 'h' : ''}
   837→        </div>

   899→async function loadEventBusData() {
   900→  try {
   901→    const [statsRes, eventsRes, rulesRes] = await Promise.all([
   902→      fetch(API + '/api/v1/events/stats'),
   903→      fetch(API + '/api/v1/events?limit=30'),
   904→      fetch(API + '/api/v1/reactions/rules?include_disabled=true'),
   905→    ]);
   906→    const stats = await statsRes.json();
   907→    const events = await eventsRes.json();
   908→    const rules = await rulesRes.json();
   909→
   910→    // Stats row
   911→    document.getElementById('ebStatsRow').innerHTML = `
   912→      <div class="stat-card"><div class="stat-label">Total Events</div><div class="stat-value info">${stats.total_events}</div></div>
   913→      <div class="stat-card"><div class="stat-label">Published</div><div class="stat-value ok">${stats.events_published}</div></div>
   914→      <div class="stat-card"><div class="stat-label">Processed</div><div class="stat-value ok">${stats.events_processed}</div></div>
   915→      <div class="stat-card"><div class="stat-label">Queue Depth</div><div class="stat-value high">${stats.queue_depth}</div></div>
   916→      <div class="stat-card"><div class="stat-label">Transport</div><div class="stat-value" style="font-size:16px; color:var(--cyan)">${stats.transport}</div></div>
   917→      <div class="stat-card"><div class="stat-label">Running</div><div class="stat-value ${stats.running ? 'ok' : 'high'}">${stats.running ? 'Yes' : 'No'}</div></div>
   918→    `;
   919→
   920→    // Events table
   921→    if (events.length === 0) {
   922→      document.getElementById('ebPanelEvents').innerHTML = '<div class="empty">No events yet. Run an audit or publish a test event.</div>';
   923→    } else {
   924→      let html = '<table><tr><th>Type</th><th>Source</th><th>Correlation ID</th><th>Timestamp</th><th>Payload (key fields)</th></tr>';
   925→      for (const e of events) {
   926→        const payloadKeys = Object.entries(e.payload || {}).slice(0, 3).map(([k,v]) => `${k}: ${typeof v === 'string' ? v : JSON.stringify(v)}`).join(' | ');
   927→        const isReaction = e.event_type === 'reaction.triggered';
   928→        html += `<tr>
   929→          <td><span class="evt-tag ${isReaction ? 'reaction' : ''}">${e.event_type}</span></td>
   930→          <td style="font-size:12px">${e.source}</td>
   931→          <td style="font-size:11px; color:var(--text-dim); font-family:monospace">${(e.correlation_id || '').slice(0,12)}...</td>
   932→          <td style="font-size:11px; color:var(--text-dim)">${(e.timestamp || '').slice(0,19)}</td>
   933→          <td style="font-size:11px; max-width:300px; overflow:hidden; text-overflow:ellipsis; white-space:nowrap">${payloadKeys}</td>
   934→        </tr>`;
   935→      }
   936→      html += '</table>';
   937→      document.getElementById('ebPanelEvents').innerHTML = html;
   938→    }
   939→
   940→    // Rules
   941→    let rulesHtml = '';
   942→    for (const rule of rules) {
   943→      rulesHtml += `
   944→        <div class="rule-card">
   945→          <div class="rule-priority">P${rule.priority}</div>
   946→          <div class="rule-info">
   947→            <div class="rule-name">${rule.name} ${rule.is_builtin ? '<span style="font-size:10px;color:var(--text-dim)">[builtin]</span>' : ''}</div>
   948→            <div class="rule-desc">${rule.description}</div>
   949→            <div class="rule-count">Events: ${rule.event_types.join(', ')} | Fired: ${rule.action_count}x | Last: ${rule.last_triggered ? rule.last_triggered.slice(0,19) : 'never'}</div>
   950→          </div>
   951→          <label class="toggle-switch">
   952→            <input type="checkbox" ${rule.enabled ? 'checked' : ''} onchange="toggleRule('${rule.rule_id}', this.checked)">
   953→            <span class="toggle-slider"></span>
   954→          </label>
   955→        </div>
   956→      `;
   957→    }
   958→    document.getElementById('ebPanelRules').innerHTML = rulesHtml || '<div class="empty">No reaction rules loaded</div>';
   959→
   960→    toast(`Event bus: ${stats.total_events} events, ${rules.length} rules`, 'success');
   961→  } catch(e) {
   962→    toast('Event bus load error: ' + e.message, 'error');
   963→  }
   964→}
