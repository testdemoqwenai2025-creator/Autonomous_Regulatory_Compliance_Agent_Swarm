   234→   234→// ── Main Dashboard ──────────────────────────────────────────────
   235→   235→
   236→   236→export default function ComplianceDashboard() {
   237→   237→  const [findings, setFindings] = useState<Finding[]>([]);
   238→   238→  const [mttr, setMttr] = useState<MTTRData | null>(null);
   239→   239→  const [policies, setPolicies] = useState<Policy[]>([]);
   240→   240→  const [profiles, setProfiles] = useState<Profile[]>([]);
   241→   241→  const [loading, setLoading] = useState(true);
   242→   242→  const [auditing, setAuditing] = useState(false);
   243→   243→
   244→   244→  // Pipeline state
   245→   245→  const [pingData, setPingData] = useState<PingData | null>(null);
   246→   246→  const [pingLoading, setPingLoading] = useState(false);
   247→   247→  const [pingError, setPingError] = useState<string | null>(null);
   248→   248→  const [postPayload, setPostPayload] = useState('');
   249→   249→  const [postResult, setPostResult] = useState<string | null>(null);
   250→   250→  const [responseHeaders, setResponseHeaders] = useState<Record<string, string>>({});
   251→   251→
   252→   252→  const fetchData = useCallback(async () => {
   253→   253→    setLoading(true);
   254→   254→    try {
   255→   255→      const [fRes, mRes, pRes, prRes] = await Promise.all([
   256→   256→        fetch('/api/compliance/findings'),
   257→   257→        fetch('/api/compliance/mttr'),
   258→   258→        fetch('/api/compliance/policies'),
   259→   259→        fetch('/api/compliance/profiles'),
   260→   260→      ]);
   261→   261→      const fData = await fRes.json();
   262→   262→      const mData = await mRes.json();
   263→   263→      const pData = await pRes.json();
   264→   264→      const prData = await prRes.json();
   265→   265→      setFindings(fData.findings);
   266→   266→      setMttr(mData);
   267→   267→      setPolicies(pData.policies);
   268→   268→      setProfiles(prData.profiles);
   269→   269→    } catch (e) {
   270→   270→      console.error('Failed to fetch compliance data:', e);
   271→   271→    } finally {
   272→   272→      setLoading(false);
   273→   273→    }
   274→   274→  }, []);
   275→   275→
   276→   276→  useEffect(() => { fetchData(); }, [fetchData]);
   277→   277→
   278→   278→  const runAudit = async () => {
   279→   279→    setAuditing(true);
   280→   280→    try {
   281→   281→      const res = await fetch('/api/compliance/audit', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({}) });
   282→   282→      const data = await res.json();
   283→   283→      if (data.status === 'audit_complete') await fetchData();
   284→   284→    } catch (e) { console.error(e); }
   285→   285→    finally { setAuditing(false); }
   286→   286→  };
   287→   287→
   288→   288→  // ── Pipeline: call /api/system/ping (GET) ──
   289→   289→  const runPing = useCallback(async () => {
   290→   290→    setPingLoading(true);
   291→   291→    setPingError(null);
   292→   292→    setPostResult(null);
   293→   293→    try {
   294→   294→      const res = await fetch('/api/system/ping');
   295→   295→      // Capture response headers to prove middleware set them
   296→   296→      const headers: Record<string, string> = {};
   297→   297→      res.headers.forEach((v, k) => { headers[k] = v; });
   298→   298→      setResponseHeaders(headers);
   299→   299→      const data: PingData = await res.json();
   300→   300→      setPingData(data);
   301→   301→    } catch (e) {
   302→   302→      setPingError(e instanceof Error ? e.message : 'Request failed');
   303→   303→    } finally {
   304→   304→      setPingLoading(false);
   305→   305→    }
   306→   306→  }, []);
   307→   307→
   308→   308→  // ── Pipeline: call /api/system/ping (POST) ──
   309→   309→  const runPostPing = useCallback(async () => {
   310→   310→    setPingLoading(true);
   311→   311→    setPingError(null);
   312→   312→    setPostResult(null);
   313→   313→    try {
   314→   314→      const payload = postPayload.trim() ? JSON.parse(postPayload) : { test: true, timestamp: new Date().toISOString() };
   315→   315→      const res = await fetch('/api/system/ping', {
   316→   316→        method: 'POST',
   317→   317→        headers: { 'Content-Type': 'application/json' },
   318→   318→        body: JSON.stringify(payload),
   319→   319→      });
   320→   320→      const data = await res.json();
   321→   321→      setPostResult(JSON.stringify(data, null, 2));
   322→   322→      // Also refresh GET data to see the new event
   323→   323→      await runPing();
   324→   324→    } catch (e) {
   325→   325→      setPingError(e instanceof Error ? e.message : 'POST failed');
   326→   326→    } finally {
   327→   327→      setPingLoading(false);
   328→   328→    }
   329→   329→  }, [postPayload, runPing]);
   330→   330→
   331→   331→  const overall = mttr?.report.overall;
   332→   332→  const openCritical = findings.filter(f => f.severity === 'critical' && (f.status === 'open' || f.status === 'in_progress')).length;
   333→   333→
   334→   334→  return (
   335→   335→    <div className="min-h-screen bg-slate-50">
   336→   336→      {/* Header */}
   337→   337→      <header className="sticky top-0 z-40 border-b border-slate-200 bg-white/80 backdrop-blur-md">
   338→   338→        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   339→   339→          <div className="flex h-16 items-center justify-between">
   340→   340→            <div className="flex items-center gap-3">
   341→   341→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-slate-900">
   342→   342→                <Shield className="h-5 w-5 text-white" />
   343→   343→              </div>
   344→   344→              <div>
   345→   345→                <h1 className="text-lg font-bold text-slate-900 leading-tight">Maritime Compliance Swarm</h1>
   346→   346→                <p className="text-xs text-slate-500">GDPR / PII / EDI Governance Platform</p>
   347→   347→              </div>
   348→   348→            </div>
   349→   349→            <div className="flex items-center gap-2">
   350→   350→              <Button variant="outline" size="sm" onClick={fetchData} disabled={loading}>
   351→   351→                <RefreshCw className={`mr-1.5 h-3.5 w-3.5 ${loading ? 'animate-spin' : ''}`} />
   352→   352→                Refresh
   353→   353→              </Button>
   354→   354→              <Button size="sm" onClick={runAudit} disabled={auditing}>
   355→   355→                {auditing ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <FileSearch className="mr-1.5 h-3.5 w-3.5" />}
   356→   356→                Run Audit
   357→   357→              </Button>
   358→   358→            </div>
   359→   359→          </div>
   360→   360→        </div>
   361→   361→      </header>
   362→   362→
   363→   363→      <main className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-6 space-y-6">
   364→   364→        {/* KPI Row */}
   365→   365→        {loading ? (
   366→   366→          <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">{Array.from({ length: 4 }).map((_, i) => <KPISkeleton key={i} />)}</div>
   367→   367→        ) : (
   368→   368→          <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   369→   369→            <KPICard icon={ShieldAlert} label="Open Findings" value={overall?.open_findings ?? '-'} sub={`${openCritical} critical`} color={openCritical > 0 ? 'text-red-700' : 'text-slate-900'} />
   370→   370→            <KPICard icon={ShieldCheck} label="Compliance Rate" value={`${overall?.compliance_rate ?? 0}%`} sub={`${overall?.resolved_findings ?? 0} resolved`} color={(overall?.compliance_rate ?? 0) >= 80 ? 'text-emerald-700' : 'text-amber-700'} />
   371→   371→            <KPICard icon={Clock} label="Avg MTTR" value={`${overall?.avg_mttr_hours ?? 0}h`} sub={`P95: ${overall?.p95_mttr_hours ?? 0}h`} />
   372→   372→            <KPICard icon={Database} label="Active Policies" value={policies.filter(p => p.enabled).length} sub={`${policies.length} total`} />
   373→   373→          </div>
   374→   374→        )}
   375→   375→
   376→   376→        {/* Tool Status Bar */}
   377→   377→        <Card>
   378→   378→          <CardContent className="p-4">
   379→   379→            <div className="flex flex-wrap items-center gap-4">
   380→   380→              <div className="flex items-center gap-2 text-sm"><Server className="w-4 h-4 text-slate-400" /><span className="font-medium text-slate-700">Swarm Tools:</span></div>
   381→   381→              {[
   382→   382→                { name: 'PII Anonymiser', icon: FileKey, lang: 'Python' },
   383→   383→                { name: 'EDI Auditor', icon: FileSearch, lang: 'Python' },
   384→   384→                { name: 'Remediation', icon: Wrench, lang: 'Python' },
   385→   385→                { name: 'MTTR Tracker', icon: Timer, lang: 'Go' },
   386→   386→              ].map(tool => (
   387→   387→                <Badge key={tool.name} variant="outline" className="gap-1.5 font-mono text-xs">
   388→   388→                  <tool.icon className="w-3 h-3" />
   389→   389→                  {tool.name}
   390→   390→                  <span className="text-slate-400">({tool.lang})</span>
   391→   391→                </Badge>
   392→   392→              ))}
   393→   393→            </div>
   394→   394→          </CardContent>
   395→   395→        </Card>
   396→   396→
   397→   397→        {/* Tabs */}
   398→   398→        <Tabs defaultValue="pipeline" className="space-y-4">
   399→   399→          <TabsList className="grid w-full grid-cols-6">
   400→   400→            <TabsTrigger value="pipeline" className="text-xs sm:text-sm">Pipeline</TabsTrigger>
   401→   401→            <TabsTrigger value="findings" className="text-xs sm:text-sm">Findings</TabsTrigger>
   402→   402→            <TabsTrigger value="mttr" className="text-xs sm:text-sm">MTTR</TabsTrigger>
   403→   403→            <TabsTrigger value="profiles" className="text-xs sm:text-sm">EDI Profiles</TabsTrigger>
   404→   404→            <TabsTrigger value="policies" className="text-xs sm:text-sm">Policies</TabsTrigger>
   405→   405→            <TabsTrigger value="endpoints" className="text-xs sm:text-sm">API</TabsTrigger>
   406→   406→          </TabsList>
   407→   407→
   408→   408→          {/* ═══════════════════════ PIPELINE TAB ═══════════════════════ */}
   409→   409→          <TabsContent value="pipeline" className="space-y-4">
   410→   410→            {/* Hero: flow diagram */}
   411→   411→            <Card>
   412→   412→              <CardHeader className="pb-3">
   413→   413→                <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   414→   414→                  <div>
   415→   415→                    <CardTitle className="text-base flex items-center gap-2">
   416→   416→                      <CircuitBoard className="w-4 h-4" />
   417→   417→                      Endpoint Communication Pipeline
   418→   418→                    </CardTitle>
   419→   419→                    <CardDescription>Real-time proof: Browser → Middleware → API Handler → Database → Response</CardDescription>
   420→   420→                  </div>
   421→   421→                  <div className="flex gap-2">
   422→   422→                    <Button size="sm" onClick={runPing} disabled={pingLoading}>
   423→   423→                      {pingLoading ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <Zap className="mr-1.5 h-3.5 w-3.5" />}
   424→   424→                      Send GET /api/system/ping
   425→   425→                    </Button>
   426→   426→                  </div>
