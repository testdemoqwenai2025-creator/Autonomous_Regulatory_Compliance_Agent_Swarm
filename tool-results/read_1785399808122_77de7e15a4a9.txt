   130→# ---------------------------------------------------------------------------
   131→# Abstract Data Ingesters
   132→# ---------------------------------------------------------------------------
   133→
   134→class DataIngester(ABC):
   135→    """Abstract base for all data ingestion sources."""
   136→
   137→    def __init__(self, source_name: str):
   138→        self.source_name = source_name
   139→        self._running = False
   140→        self._processed_count = 0
   141→        self._error_count = 0
   142→        self._last_ingest: Optional[datetime] = None
   143→
   144→    @abstractmethod
   145→    def connect(self) -> bool:
   146→        """Establish connection to the data source. Returns True on success."""
   147→        ...
   148→
   149→    @abstractmethod
   150→    def ingest(self) -> list[Any]:
   151→        """Ingest and return new data points. Returns empty list if no new data."""
   152→        ...
   153→
   154→    def disconnect(self) -> None:
   155→        """Clean up connections."""
   156→        self._running = False
   157→
   158→    @property
   159→    def stats(self) -> dict:
   160→        return {
   161→            "source": self.source_name,
   162→            "running": self._running,
   163→            "processed_count": self._processed_count,
   164→            "error_count": self._error_count,
   165→            "last_ingest": self._last_ingest.isoformat() if self._last_ingest else None,
   166→        }

   269→# ---------------------------------------------------------------------------
   270→# Compliance Correlation Engine
   271→# ---------------------------------------------------------------------------
   272→
   273→class ComplianceCorrelator:
   274→    """Correlates external data (weather, AIS, emissions) with compliance findings.
   275→
   276→    Produces ComplianceCorrelationEvents that can be published to the
   277→    event bus for downstream reaction processing.
   278→    """
   279→
   280→    def __init__(self):
   281→        self._correlation_rules: list[Callable] = [
   282→            self._check_weather_hold,
   283→            self._check_route_deviation,
   284→            self._check_emissions_breach,
   285→            self._check_port_closure,
   286→        ]
   287→
   288→    def correlate(self, finding: object, external_data: list[Any]) -> list[ComplianceCorrelationEvent]:
   289→        """Run all correlation rules against a finding and external data."""
   290→        events = []
   291→        for rule in self._correlation_rules:
   292→            try:
   293→                result = rule(finding, external_data)
   294→                if result:
   295→                    events.extend(result if isinstance(result, list) else [result])
   296→            except Exception as e:
   297→                logger.error(f"Correlation rule error: {e}")
   298→        return events
   299→
   300→    def _check_weather_hold(self, finding: object, data: list[Any]) -> Optional[ComplianceCorrelationEvent]:
   301→        """Check if active weather event warrants SLA pause."""
   302→        weather_data = [d for d in data if isinstance(d, WeatherObservation)]
   303→        for obs in weather_data:
   304→            if obs.wave_height_m > 4.0 or obs.wind_speed_knots > 50:
   305→                return ComplianceCorrelationEvent(
   306→                    event_type="weather_hold",
   307→                    finding_id=getattr(finding, "id", None),
   308→                    correlation_type="weather",
   309→                    severity_impact="downgraded",
   310→                    description=f"Severe weather detected: wind {obs.wind_speed_knots}kn, "
   311→                                f"waves {obs.wave_height_m}m. SLA clock paused.",
   312→                    external_data_ref=f"weather:{obs.observation_id}",
   313→                    confidence=0.85,
   314→                )
   315→        return None
   316→
   317→    def _check_route_deviation(self, finding: object, data: list[Any]) -> Optional[ComplianceCorrelationEvent]:
   318→        """Check if AIS data shows unexpected route deviation."""
   319→        ais_data = [d for d in data if isinstance(d, AISPositionReport)]
   320→        if len(ais_data) >= 2:
   321→            latest = ais_data[-1]
   322→            if latest.navigation_status == NavigationStatus.NOT_UNDER_COMMAND:
   323→                return ComplianceCorrelationEvent(
   324→                    event_type="route_deviation",
   325→                    vessel_id=latest.mmsi,
   326→                    correlation_type="route",
   327→                    severity_impact="upgraded",
   328→                    description=f"Vessel {latest.mmsi} not under command at "
   329→                                f"({latest.latitude:.4f}, {latest.longitude:.4f}). "
   330→                                f"Compliance risk upgraded.",
   331→                    external_data_ref=f"ais:{latest.mmsi}",
   332→                    confidence=0.90,
   333→                )
   334→        return None
   335→
   336→    def _check_emissions_breach(self, finding: object, data: list[Any]) -> Optional[ComplianceCorrelationEvent]:
   337→        """Check if emissions readings exceed EU ETS / IMO thresholds."""
   338→        emissions_data = [d for d in data if isinstance(d, EmissionsReading)]
   339→        for reading in emissions_data:
   340→            if reading.sox_emissions_kg > 500 or reading.nox_emissions_kg > 2000:
   341→                return ComplianceCorrelationEvent(
   342→                    event_type="emissions_breach",
   343→                    vessel_id=reading.vessel_id,
   344→                    correlation_type="emissions",
   345→                    severity_impact="upgraded",
   346→                    description=f"Vessel {reading.vessel_id} emissions breach: "
   347→                                f"SOx {reading.sox_emissions_kg}kg, "
   348→                                f"NOx {reading.nox_emissions_kg}kg.",
   349→                    external_data_ref=f"emissions:{reading.vessel_id}",
   350→                    confidence=0.95,
   351→                )
   352→        return None
   353→
   354→    def _check_port_closure(self, finding: object, data: list[Any]) -> Optional[ComplianceCorrelationEvent]:
   355→        """Check if port closure events affect finding timelines."""
   356→        # Port closures would come through as weather events or explicit notifications
   357→        weather_data = [d for d in data if isinstance(d, WeatherObservation)]
   358→        for obs in weather_data:
   359→            if obs.visibility_nm < 0.5 and obs.wind_speed_knots > 65:
   360→                return ComplianceCorrelationEvent(
   361→                    event_type="port_closure",
   362→                    finding_id=getattr(finding, "id", None),
   363→                    correlation_type="weather",
   364→                    severity_impact="downgraded",
   365→                    description=f"Port closure conditions detected: visibility {obs.visibility_nm}nm, "
   366→                                f"wind {obs.wind_speed_knots}kn. Finding SLA extended.",
   367→                    external_data_ref=f"weather:{obs.observation_id}",
   368→                    confidence=0.75,
   369→                )
   370→        return None

   373→# ---------------------------------------------------------------------------
   374→# Data Pipeline Orchestrator
   375→# ---------------------------------------------------------------------------
   376→
   377→class DataPipelineOrchestrator:
   378→    """Orchestrates data ingestion from multiple external sources.
   379→
   380→    In production, runs as a background service that:
   381→    1. Polls/connects to each data source
   382→    2. Normalises incoming data
   383→    3. Runs compliance correlation
   384→    4. Publishes correlation events to the event bus
   385→    """
   386→
   387→    def __init__(self):
   388→        self._ingesters: dict[str, DataIngester] = {}
   389→        self._correlator = ComplianceCorrelator()
   390→        self._event_callback: Optional[Callable] = None
   391→        self._running = False
   392→
   393→    def register_ingester(self, name: str, ingester: DataIngester) -> None:
   394→        self._ingesters[name] = ingester
   395→
   396→    def set_event_callback(self, callback: Callable) -> None:
   397→        """Set callback for publishing correlation events to the event bus."""
   398→        self._event_callback = callback
   399→
   400→    async def run_cycle(self) -> dict:
   401→        """Execute one ingestion cycle across all sources. Returns stats."""
   402→        if not self._running:
   403→            return {"status": "not_running"}
   404→
   405→        cycle_stats = {}
   406→        all_data = []
   407→
   408→        for name, ingester in self._ingesters.items():
   409→            try:
   410→                data = ingester.ingest()
   411→                all_data.extend(data)
   412→                cycle_stats[name] = {
   413→                    "status": "ok",
   414→                    "records": len(data),
   415→                }
   416→            except Exception as e:
   417→                logger.error(f"Ingestion error for {name}: {e}")
   418→                cycle_stats[name] = {
   419→                    "status": "error",
   420→                    "error": str(e),
   421→                }
   422→
   423→        return {
   424→            "status": "completed",
   425→            "timestamp": datetime.now(timezone.utc).isoformat(),
   426→            "sources": cycle_stats,
   427→            "total_records": len(all_data),
   428→        }
   429→
   430→    def start(self) -> None:
   431→        """Start all ingesters."""
   432→        self._running = True
   433→        for name, ingester in self._ingesters.items():
   434→            ingester.connect()
   435→        logger.info(f"Data pipeline started with {len(self._ingesters)} sources")
   436→
   437→    def stop(self) -> None:
   438→        """Stop all ingesters."""
   439→        self._running = False
   440→        for ingester in self._ingesters.values():
   441→            ingester.disconnect()
   442→        logger.info("Data pipeline stopped")
   443→
   444→    @property
   445→    def stats(self) -> dict:
   446→        return {
   447→            "running": self._running,
   448→            "sources": {name: ingester.stats for name, ingester in self._ingesters.items()},
   449→            "correlator": "active" if self._running else "stopped",
   450→        }

   552→
   553→def ais_decode(raw_payloads: list[dict]) -> list[AISMessage]:
   554→    """Pipeline stage: Decode raw provider payloads into AISMessage objects.
   555→
   556→    In production, this parses NMEA VDM sentences or provider-specific JSON.
   557→    For development, it extracts fields from pre-parsed dicts.
   558→    """
   559→    messages = []
   560→    for payload in raw_payloads:
   561→        try:
   562→            ts = payload.get("timestamp")
   563→            if isinstance(ts, str):
   564→                ts = datetime.fromisoformat(ts.replace("Z", "+00:00"))
   565→            elif ts is None:
   566→                ts = datetime.now(timezone.utc)
   567→            messages.append(AISMessage(
   568→                mmsi=str(payload.get("mmsi", "")),
   569→                lat=float(payload.get("latitude", 0.0)),
   570→                lon=float(payload.get("longitude", 0.0)),
   571→                heading=float(payload.get("heading", 0.0)),
   572→                speed=float(payload.get("speed", 0.0)),
   573→                timestamp=ts,
   574→                nav_status=str(payload.get("nav_status", "")),
   575→                message_type=int(payload.get("message_type", 1)),
   576→            ))
   577→        except (ValueError, TypeError) as e:
   578→            logger.warning(f"AIS decode error: {e}")
   579→    return messages

   582→def ais_deduplicate(messages: list[AISMessage], window_seconds: float = 300.0) -> list[AISMessage]:
   583→    """Pipeline stage: Remove duplicate messages within a time window.
   584→
   585→    Duplicates are identified by same MMSI with timestamps within
   586→    the specified window (default 300 seconds). Keeps the most
   587→    recent message in each dedup group.
   588→    """
   589→    if not messages:
   590→        return []
   591→
   592→    messages.sort(key=lambda m: (m.mmsi, m.timestamp))
   593→    result: list[AISMessage] = []
   594→    seen: dict[str, datetime] = {}
   595→
   596→    for msg in messages:
   597→        last_ts = seen.get(msg.mmsi)
   598→        if last_ts is None:
   599→            seen[msg.mmsi] = msg.timestamp
   600→            result.append(msg)
   601→        else:
   602→            delta = (msg.timestamp - last_ts).total_seconds()
   603→            if delta > window_seconds:
   604→                seen[msg.mmsi] = msg.timestamp
   605→                result.append(msg)
   606→            # else: duplicate within window, skip
   607→
   608→    logger.debug(f"AIS dedup: {len(messages)} -> {len(result)} messages (window={window_seconds}s)")
   609→    return result
   610→

   612→def ais_temporal_interpolate(
   613→    messages: list[AISMessage],
   614→    max_gap_minutes: float = 30.0,
   615→) -> list[AISMessage]:
   616→    """Pipeline stage: Linear interpolation for gaps shorter than max_gap_minutes.
   617→
   618→    For each MMSI, if the time gap between consecutive messages is less than
   619→    max_gap_minutes, insert interpolated points at 5-minute intervals.
   620→    This improves route continuity for compliance checks.
   621→    """
   622→    if not messages:
   623→        return []
   624→
   625→    max_gap_seconds = max_gap_minutes * 60.0
   626→    interval_seconds = 300.0  # 5-minute intervals
   627→
   628→    by_mmsi: dict[str, list[AISMessage]] = defaultdict(list)
   629→    for msg in messages:
   630→        by_mmsi[msg.mmsi].append(msg)
   631→
   632→    result: list[AISMessage] = []
   633→    for mmsi, track in by_mmsi.items():
   634→        track.sort(key=lambda m: m.timestamp)
   635→        result.append(track[0])
   636→        for i in range(1, len(track)):
   637→            prev, curr = track[i - 1], track[i]
   638→            gap = (curr.timestamp - prev.timestamp).total_seconds()
   639→            if 0 < gap < max_gap_seconds:
   640→                n_points = int(gap / interval_seconds)
   641→                for j in range(1, n_points):
   642→                    frac = (j * interval_seconds) / gap
   643→                    interp_lat = prev.lat + frac * (curr.lat - prev.lat)
   644→                    interp_lon = prev.lon + frac * (curr.lon - prev.lon)
   645→                    interp_heading = prev.heading + frac * (curr.heading - prev.heading)
   646→                    interp_speed = prev.speed + frac * (curr.speed - prev.speed)
   647→                    interp_ts = prev.timestamp + __import__("datetime").timedelta(seconds=j * interval_seconds)
   648→                    result.append(AISMessage(
   649→                        mmsi=mmsi,
   650→                        lat=interp_lat,
   651→                        lon=interp_lon,
   652→                        heading=interp_heading % 360,
   653→                        speed=max(0.0, interp_speed),
   654→                        timestamp=interp_ts,
   655→                        nav_status=prev.nav_status,
   656→                        message_type=prev.message_type,
   657→                    ))
   658→            result.append(curr)
   659→
   660→    logger.debug(f"AIS interpolation: {len(messages)} -> {len(result)} messages")
   661→    return result
   662→

   664→def ais_spatial_index(messages: list[AISMessage], target: str = "postgis") -> list[dict]:
   665→    """Pipeline stage: Create spatial index entries.
   666→
   667→    For PostGIS target, generates ST_MakePoint SQL expressions.
   668→    For memory target, returns dicts with lat/lon pairs ready for
   669→    in-memory spatial indexing (e.g., R-tree).
   670→    """
   671→    entries = []
   672→    for msg in messages:
   673→        if target == "postgis":
   674→            entries.append({
   675→                "mmsi": msg.mmsi,
   676→                "timestamp": msg.timestamp.isoformat(),
   677→                "geom_sql": f"ST_MakePoint({msg.lon}, {msg.lat}, 4326)",
   678→                "heading": msg.heading,
   679→                "speed": msg.speed,
   680→            })
   681→        else:
   682→            entries.append({
   683→                "mmsi": msg.mmsi,
   684→                "timestamp": msg.timestamp.isoformat(),
   685→                "lat": msg.lat,
   686→                "lon": msg.lon,
   687→                "heading": msg.heading,
   688→                "speed": msg.speed,
   689→            })
   690→    return entries

   693→def generate_compliance_finding(
   694→    event_type: str,
   695→    mmsi: str,
   696→    description: str,
   697→    severity: str = "medium",
   698→    confidence: float = 0.8,
   699→    evidence: Optional[dict] = None,
   700→) -> dict:
   701→    """Create a compliance finding dict ready for the event bus.
   702→
   703→    This stub produces a dict conforming to the event bus schema.
   704→    Downstream consumers (compliance engine, risk scorer) can
   705→    process these findings immediately.
   706→    """
   707→    finding = {
   708→        "event_type": event_type,
   709→        "source": "satellite_ais_pipeline",
   710→        "mmsi": mmsi,
   711→        "description": description,
   712→        "severity": severity,
   713→        "confidence": confidence,
   714→        "timestamp": datetime.now(timezone.utc).isoformat(),
   715→        "evidence": evidence or {},
   716→    }
   717→    if event_type not in AIS_COMPLIANCE_EVENT_TYPES:
   718→        logger.warning(f"Unknown AIS compliance event type: {event_type}")
   719→    return finding
