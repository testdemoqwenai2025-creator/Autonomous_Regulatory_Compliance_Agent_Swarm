   831→    const flowOrder = ['detected','triaged','assigned','in_remediation','awaiting_verification','verified','closed'];

  1254→const BS_LABELS = {
  1255→  database: 'Database', state_machine: 'State Machine', event_bus: 'Event Bus',
  1256→  reaction_engine: 'Reaction Engine', anonymiser: 'PII Anonymiser', ner_detector: 'NER Detector',
  1257→  auditor: 'EDI Auditor', remediation: 'Remediation Generator', mttr_tracker: 'MTTR Tracker (Go)',
  1258→  event_flow: 'End-to-End Event Flow'
  1259→};
  1260→
  1261→async function loadBackendStatus() {
  1262→  try {
  1263→    const r = await fetch(API + '/api/v1/system/frontend-status');
  1264→    const d = await r.json();
  1265→
  1266→    // Update banner
  1267→    const banner = document.getElementById('bsBanner');
  1268→    banner.className = 'backend-status-banner ' + (d.status === 'operational' ? 'operational' : d.status === 'degraded' ? 'degraded' : 'down');
  1269→    document.getElementById('bsBannerText').textContent = d.status === 'operational'
  1270→      ? 'All backend services operational — frontend communicating successfully'
  1271→      : 'Some backend services degraded';
  1272→
  1273→    const services = d.services || {};
  1274→    const okCount = Object.values(services).filter(s => s.status === 'ok').length;
  1275→    const totalCount = Object.keys(services).length;
  1276→    document.getElementById('bsBannerCount').textContent = `${okCount} / ${totalCount} services OK`;
  1277→
  1278→    // Update header status dot + text
  1279→    const dot = document.getElementById('statusDot');
  1280→    const txt = document.getElementById('statusText');
  1281→    if (d.status === 'operational') {
  1282→      dot.className = 'status-dot';
  1283→      txt.textContent = `${okCount}/${totalCount} services connected`;
  1284→    } else {
  1285→      dot.className = 'status-dot degraded';
  1286→      txt.textContent = `${okCount}/${totalCount} services — degraded`;
  1287→    }
  1288→
  1289→    // Build service cards
  1290→    let html = '';
  1291→    for (const [key, svc] of Object.entries(services)) {
  1292→      const icon = BS_ICONS[key] || '\u{1F4CC}';
  1293→      const label = BS_LABELS[key] || key;
  1294→      const statusCls = svc.status || 'error';
  1295→      html += `
  1296→        <div class="bs-card">
  1297→          <div class="bs-icon ${statusCls}">${icon}</div>
  1298→          <div>
  1299→            <div class="bs-name">${label} <span class="badge badge-${statusCls}">${svc.status}</span></div>
  1300→            <div class="bs-detail">${svc.detail || ''}</div>
  1301→          </div>
  1302→        </div>`;
  1303→    }
  1304→    document.getElementById('bsGrid').innerHTML = html;
  1305→    document.getElementById('bsTimestamp').textContent = 'Last checked: ' + d.timestamp + '  |  Gateway v' + d.gateway_version;
  1306→  } catch(e) {
  1307→    document.getElementById('bsBanner').className = 'backend-status-banner down';
  1308→    document.getElementById('bsBannerText').textContent = 'Cannot reach backend gateway';
  1309→    document.getElementById('bsGrid').innerHTML = '<div style="color:var(--red); font-size:13px">Error: ' + e.message + '</div>';
  1310→    document.getElementById('statusDot').className = 'status-dot down';
  1311→    document.getElementById('statusText').textContent = 'Backend unreachable';
  1312→  }
  1313→}
