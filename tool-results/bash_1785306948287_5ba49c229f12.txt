     1	"""FastAPI gateway for the Maritime Global Compliance Swarm.
     2	
     3	Provides a unified REST API over all four compliance tools:
     4	  1. PII Anonymiser        - Tokenise / redact / encrypt manifest PII
     5	  2. EDI SQL Auditor       - Run compliance queries, profile checks
     6	  3. Remediation Generator - Generate masking policies, update EDI profiles
     7	  4. MTTR Tracker (Go)     - Proxy to the Golang telemetry service
     8	
     9	The gateway is a pure-Python process. It communicates with the Golang
    10	MTTR tracker over HTTP (via httpx) and with the database directly
    11	via SQLAlchemy.
    12	
    13	Usage:
    14	    uvicorn gateway.app:app --host 0.0.0.0 --port 8000
    15	"""

   152	def create_app(config: Optional[SwarmConfig] = None) -> FastAPI:
   153	    """Create and configure the FastAPI application.
   154	
   155	    Args:
   156	        config: Optional pre-built config. Loads from env if None.
   157	    """
   158	    global _config, _session_factory, _anonymiser, _rule_engine
   159	    global _auditor, _policy_generator, _edi_updater, _mttr_base_url
   160	    global _ner_detector, _query_registry, _state_machine
   161	    global _event_bus, _reaction_engine
   162	
   163	    _config = config or SwarmConfig.from_env()
   164	
   165	    # Logging
   166	    log_level = getattr(logging, _config.log_level, logging.INFO)
   167	    logging.basicConfig(
   168	        level=log_level,
   169	        format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
   170	    )
   171	
   172	    # Database
   173	    engine = create_engine_from_config(_config)
   174	    init_schema(engine)
   175	    _session_factory = get_session_factory(engine)
   176	
   177	    # Tools
   178	    _anonymiser = PIITokeniser(_config.anonymiser)
   179	    _rule_engine = RuleEngine()
   180	    _auditor = EDIAuditor(_config)
   181	    _policy_generator = PolicyGenerator(_config)
   182	    _edi_updater = EDIProfileUpdater(_config)
   183	    _ner_detector = MaritimeNERDetector()
   184	    _query_registry = QueryRegistry(_session_factory)
   185	    _query_registry.seed_defaults()
   186	
   187	    # State machine + Event bus + Reaction engine
   188	    _state_machine = create_state_machine()
   189	    is_postgres = _config.database.driver != "sqlite"
   190	    _event_bus = EventBus(_session_factory, is_postgres=is_postgres)
   191	    _reaction_engine = ReactionEngine(_event_bus, _state_machine, _session_factory)
   192	    _reaction_engine.register_builtin_rules()
   193	
   194	    # Wire state machine -> event bus bridge
   195	    # Every successful transition auto-emits an event so reactions fire immediately.
   196	    def _sm_event_bridge(transition_result):
   197	        """Callback: publish a finding.state_changed event on every successful SM transition."""
   198	        if not transition_result.success:
   199	            return
   200	        _event_bus.publish(
   201	            EventType.FINDING_STATE_CHANGED,
   202	            payload={
   203	                "finding_id": transition_result.finding_id,
   204	                "from_state": transition_result.from_state,
   205	                "to_state": transition_result.to_state,
   206	                "trigger": transition_result.trigger,
   207	                "actor": transition_result.actor,
   208	                "transition_id": transition_result.transition_id,
   209	                "auto_escalated": transition_result.auto_escalated,
   210	                "timeout_hours": transition_result.timeout_hours,
   211	                **transition_result.context,
   212	            },
   213	            source="state_machine_bridge",
   214	            correlation_id=transition_result.finding_id,
   215	        )
   216	        # Also forward to Go MTTR tracker via HTTP
   217	        _forward_sm_event_to_mttr(transition_result)
   218	
   219	    _state_machine.register_callback(_sm_event_bridge)
   220	
   221	    _event_bus.start()
   222	
   223	    # MTTR tracker URL (Golang service)
   224	    _mttr_base_url = os.getenv(
   225	        "MTTR_TRACKER_URL",
   226	        f"http://localhost:{_config.telemetry.http_port}",
   227	    )
   228	
   229	    # Module-level helper: forward state machine transitions to Go MTTR tracker
   230	    def _forward_sm_event_to_mttr(transition_result):
   231	        """Best-effort HTTP POST to the Go MTTR tracker to record a phase event.
   232	
   233	        Maps FindingState values to Go EventPhase values using the
   234	        state_machine.map_go_phase() function so the Go telemetry
   235	        service stays in sync with the Python state machine.
   236	        """
   237	        from shared.state_machine import FindingStateMachine
   238	        go_phase = FindingStateMachine.map_go_phase(transition_result.to_state)
   239	        payload = {
   240	            "finding_id": transition_result.finding_id,
   241	            "phase": go_phase,
   242	            "metadata": {
   243	                "from_state": transition_result.from_state,
   244	                "to_state": transition_result.to_state,
   245	                "trigger": transition_result.trigger,
   246	                "actor": transition_result.actor,
   247	                "transition_id": transition_result.transition_id,
   248	            },
   249	        }
   250	        if transition_result.context.get("assignee"):
   251	            payload["assignee"] = transition_result.context["assignee"]
   252	        try:
   253	            import asyncio
   254	            loop = asyncio.get_event_loop()
   255	            if loop.is_running():
   256	                asyncio.ensure_future(_async_post_mttr(payload))
   257	            else:
   258	                loop.run_until_complete(_async_post_mttr(payload))
   259	        except Exception as e:
   260	            logger.debug("MTTR tracker forward failed (non-critical): %s", e)
   261	
   262	    async def _async_post_mttr(payload):
   263	        try:
   264	            async with httpx.AsyncClient(timeout=3.0) as client:
   265	                resp = await client.post(f"{_mttr_base_url}/api/v1/events", json=payload)
   266	                if resp.status_code in (200, 201):
   267	                    logger.debug("MTTR event forwarded: phase=%s", payload["phase"])
   268	                else:
   269	                    logger.warning("MTTR tracker returned %d", resp.status_code)
   270	        except Exception as e:
   271	            logger.debug("MTTR tracker unreachable (non-critical): %s", e)
   272	
   273	    # FastAPI app
   274	    app = FastAPI(
   275	        title="Maritime Global Compliance Swarm",
   276	        description=(
   277	            "Unified API gateway for maritime data governance and privacy compliance. "
   278	            "Exposes four compliance tools: PII Anonymiser, EDI SQL Auditor, "
   279	            "Remediation Route Generator, and MTTR Telemetry Tracker."
   280	        ),
   281	        version="1.0.0",
   282	        docs_url="/docs",
   283	        redoc_url="/redoc",
   284	    )
   285	
   286	    app.add_middleware(
   287	        CORSMiddleware,
   288	        allow_origins=["*"],
   289	        allow_credentials=True,
   290	        allow_methods=["*"],
   291	        allow_headers=["*"],
   292	    )
   293	
   294	    # Serve static frontend
   295	    static_dir = Path(__file__).parent / "static"
   296	    if static_dir.is_dir():
   297	        app.mount("/static", StaticFiles(directory=str(static_dir)), name="static")
   298	
   299	        @app.get("/", include_in_schema=False, tags=["frontend"])
   300	        async def serve_frontend():
   301	            return FileResponse(str(static_dir / "index.html"))
   302	
   303	    # ── Register routers ──────────────────────────────────────────────
   304	
   305	    _register_health_routes(app)
   306	    _register_connectivity_route(app)
   307	    _register_frontend_status_route(app)
   308	    _register_anonymiser_routes(app)
   309	    _register_ner_routes(app)
   310	    _register_auditor_routes(app)
   311	    _register_registry_routes(app)
   312	    _register_remediation_routes(app)
   313	    _register_mttr_routes(app)
   314	    _register_query_routes(app)
   315	    _register_state_machine_routes(app)
   316	    _register_event_bus_routes(app)
   317	
   318	    return app

   325	def _register_health_routes(app: FastAPI):
   326	    @app.get("/health", response_model=HealthResponse, tags=["health"])
   327	    async def health_check():
   328	        """Liveness probe — returns status of all four tools."""
   329	        tool_status = {"anonymiser": "ok", "auditor": "ok", "remediation": "ok"}
   330	
   331	        # Check MTTR tracker (Golang)
   332	        try:
   333	            async with httpx.AsyncClient(timeout=3.0) as client:
   334	                resp = await client.get(f"{_mttr_base_url}/health")
   335	                tool_status["mttr_tracker"] = "ok" if resp.status_code == 200 else "degraded"
   336	        except Exception:
   337	            tool_status["mttr_tracker"] = "unreachable"
   338	
   339	        return HealthResponse(
   340	            status="healthy",
   341	            service="compliance-swarm-gateway",
   342	            tools=tool_status,
   343	        )

   350	def _register_connectivity_route(app: FastAPI):
   351	
   352	    @app.get(
   353	        "/api/v1/system/connectivity",
   354	        response_model=ConnectivityResponse,
   355	        tags=["system"],
   356	    )
   357	    async def connectivity_check():
   358	        """Comprehensive connectivity diagnostics for the frontend.
   359	
   360	        Tests every backend component and middleware connection, returning
   361	        detailed status with latency measurements. The frontend calls this
   362	        endpoint to confirm it can effectively communicate with all backend
   363	        services, tools, and the Golang MTTR tracker.
   364	
   365	        Returns per-component status: ok | degraded | unavailable.
   366	        """
   367	        from sqlalchemy import text as sa_text
   368	
   369	        ts = datetime.now(timezone.utc).isoformat()
   370	        components = {}
   371	        all_ok = True
   372	
   373	        # 1. Database connectivity
   374	        db_status = "ok"
   375	        db_latency = None
   376	        db_detail = None
   377	        try:
   378	            t0 = time.monotonic()
   379	            with _session_factory() as session:
   380	                session.execute(sa_text("SELECT 1")).scalar()
   381	            db_latency = round((time.monotonic() - t0) * 1000, 2)
   382	        except Exception as e:
   383	            db_status = "unavailable"
   384	            db_detail = str(e)[:200]
   385	            all_ok = False
   386	        database_status = ComponentStatus(
   387	            name="Compliance Database",
   388	            status=db_status,
   389	            latency_ms=db_latency,
   390	            detail=db_detail,
   391	        )
   392	        components["database"] = database_status
   393	
   394	        # 2. PII Anonymiser
   395	        t0 = time.monotonic()
   396	        try:
   397	            _anonymiser._vault.tokenise("connectivity-test", PIIFieldCategory.CONTACT_INFO)
   398	            anonymiser_status = "ok"
   399	        except Exception:
   400	            anonymiser_status = "unavailable"
   401	            all_ok = False
   402	        anonymiser_latency = round((time.monotonic() - t0) * 1000, 2)
   403	        components["anonymiser"] = ComponentStatus(
   404	            name="PII Anonymiser",
   405	            status=anonymiser_status,
   406	            latency_ms=anonymiser_latency,
   407	        )
   408	
   409	        # 3. NER Detector
   410	        t0 = time.monotonic()
   411	        try:
   412	            _ner_detector.detect("Test text")
   413	            ner_status = "ok"
   414	        except Exception:
   415	            ner_status = "unavailable"
   416	            all_ok = False
   417	        ner_latency = round((time.monotonic() - t0) * 1000, 2)
   418	        components["ner_detector"] = ComponentStatus(
   419	            name="NER Detector",
   420	            status=ner_status,
   421	            latency_ms=ner_latency,
   422	            detail=f"layers: {_ner_detector.layers_available}, spacy: {_ner_detector.spacy_available}",
   423	        )
   424	
   425	        # 4. EDI Auditor
   426	        t0 = time.monotonic()
   427	        try:
   428	            _rule_engine.find_matching_rules("consignee_name")
   429	            auditor_status = "ok"
   430	        except Exception:
   431	            auditor_status = "unavailable"
   432	            all_ok = False
   433	        auditor_latency = round((time.monotonic() - t0) * 1000, 2)
   434	        components["auditor"] = ComponentStatus(
   435	            name="EDI SQL Auditor",
   436	            status=auditor_status,
   437	            latency_ms=auditor_latency,
   438	        )
   439	
   440	        # 5. Remediation Generator
   441	        t0 = time.monotonic()
   442	        try:
   443	            from remediation.policy_gen import REMEDIATION_MATRIX
   444	            _ = len(REMEDIATION_MATRIX)
   445	            remediation_status = "ok"
   446	        except Exception:
   447	            remediation_status = "unavailable"
   448	            all_ok = False
   449	        remediation_latency = round((time.monotonic() - t0) * 1000, 2)
   450	        components["remediation"] = ComponentStatus(
   451	            name="Remediation Generator",
   452	            status=remediation_status,
   453	            latency_ms=remediation_latency,
   454	        )
   455	
   456	        # 6. MTTR Proxy (Golang)
   457	        mttr_status = "unavailable"
   458	        mttr_latency = None
   459	        mttr_detail = None
   460	        try:
   461	            t0 = time.monotonic()
   462	            async with httpx.AsyncClient(timeout=5.0) as client:
   463	                resp = await client.get(f"{_mttr_base_url}/health")
   464	                mttr_latency = round((time.monotonic() - t0) * 1000, 2)
   465	                if resp.status_code == 200:
   466	                    mttr_status = "ok"
   467	                else:
   468	                    mttr_status = "degraded"
   469	                    mttr_detail = f"HTTP {resp.status_code}"
   470	        except httpx.ConnectError:
   471	            mttr_detail = f"Cannot connect to {_mttr_base_url}"
   472	        except Exception as e:
   473	            mttr_detail = str(e)[:200]
   474	
   475	        if mttr_status == "unavailable":
   476	            all_ok = False
   477	
   478	        mttr_component = ComponentStatus(
   479	            name="MTTR Tracker (Go)",
   480	            status=mttr_status,
   481	            latency_ms=mttr_latency,
   482	            detail=mttr_detail,
   483	        )
   484	        components["mttr_tracker"] = mttr_component
   485	
   486	        # 7. Query Registry
   487	        t0 = time.monotonic()
   488	        try:
   489	            stats = _query_registry.get_statistics()
   490	            registry_status = "ok"
   491	            registry_detail = f"{stats['active_queries']} active, {stats['builtin_queries']} builtin"
   492	        except Exception as e:
   493	            registry_status = "unavailable"
   494	            registry_detail = str(e)[:200]
   495	            all_ok = False
   496	        registry_latency = round((time.monotonic() - t0) * 1000, 2)
   497	        registry_component = ComponentStatus(
   498	            name="Query Registry",
   499	            status=registry_status,
   500	            latency_ms=registry_latency,
   501	            detail=registry_detail,
   502	        )
   503	        components["query_registry"] = registry_component
   504	
   505	        overall = "ok" if all_ok else ("degraded" if db_status == "ok" else "down")
   506	
   507	        # Count routes
   508	        total_routes = len(app.routes)
   509	        active_routes = sum(
   510	            1 for r in app.routes
   511	            if hasattr(r, "methods") and hasattr(r, "path")
   512	        )
   513	
   514	        # 8. State Machine
   515	        t0 = time.monotonic()
   516	        try:
   517	            _state_machine.get_full_definition()
   518	            sm_status = "ok"
   519	        except Exception:
   520	            sm_status = "unavailable"
   521	            all_ok = False
   522	        sm_latency = round((time.monotonic() - t0) * 1000, 2)
   523	        sm_component = ComponentStatus(
   524	            name="State Machine",
   525	            status=sm_status,
   526	            latency_ms=sm_latency,
   527	            detail=f"{len(_state_machine.get_full_definition()['states'])} states, {_state_machine.get_full_definition()['total_transitions']} transitions",
   528	        )
   529	        components["state_machine"] = sm_component
   530	
   531	        # 9. Event Bus
   532	        t0 = time.monotonic()
   533	        try:
   534	            eb_stats = _event_bus.get_statistics()
   535	            eb_status = "ok" if eb_stats["running"] else "degraded"
   536	        except Exception as e:
   537	            eb_status = "unavailable"
   538	            all_ok = False
   539	            eb_stats = {}
   540	        eb_latency = round((time.monotonic() - t0) * 1000, 2)
   541	        eb_component = ComponentStatus(
   542	            name="Event Bus",
   543	            status=eb_status,
   544	            latency_ms=eb_latency,
   545	            detail=f"{eb_stats.get('transport', 'unknown')}, {eb_stats.get('total_events', 0)} events stored",
   546	        )
   547	        components["event_bus"] = eb_component
   548	
   549	        # 10. Reaction Engine
   550	        t0 = time.monotonic()
   551	        try:
   552	            re_stats = _reaction_engine.get_statistics()
   553	            re_status = "ok"
   554	        except Exception:
   555	            re_status = "unavailable"
   556	            all_ok = False
   557	            re_stats = {}
   558	        re_latency = round((time.monotonic() - t0) * 1000, 2)
   559	        re_component = ComponentStatus(
   560	            name="Reaction Engine",
   561	            status=re_status,
   562	            latency_ms=re_latency,
   563	            detail=f"{re_stats.get('enabled_rules', 0)} active rules, {re_stats.get('total_actions_executed', 0)} actions fired",
   564	        )
   565	        components["reaction_engine"] = re_component
   566	
   567	        return ConnectivityResponse(
   568	            status=overall,
   569	            timestamp=ts,
   570	            gateway_version="2.0.0",
   571	            components=components,
   572	            database=database_status,
   573	            ner_layers=_ner_detector.layers_available,
   574	            mttr_proxy=mttr_component,
   575	            query_registry=registry_component,
   576	            state_machine=sm_component,
   577	            event_bus=eb_component,
   578	            reaction_engine=re_component,
   579	            active_routes=active_routes,
   580	            total_routes=total_routes,
   581	        )

   600	        endpoint returns a simple summary designed for the UI status bar.

   604	          - State machine is operational (definition loaded)
   605	          - Event bus is running and can publish/consume

   613	        # 1. Database

   616	            with _session_factory() as session:
   617	                session.execute(sa_text("SELECT 1")).scalar()
   618	            services["database"] = {"status": "ok", "detail": "read/write verified"}

   640	                "events_stored": eb_stats.get("total_events", 0),
   641	                "subscribers": eb_stats.get("subscriber_count", 0),

   643	            }

   669	            services["ner_detector"] = {

   681	        except Exception as e:

   683

  1715	                    "closed": AuditStatus.REMEDIATED,
  1716	                    "false_positive": AuditStatus.FALSE_POSITIVE,
  1717	                }
  1718	                finding.status = state_to_status.get(req.target_state.value, AuditStatus.OPEN)
  1719

  1805	    async def sm_check_timeouts():
  1806	        """Check all open findings for timeout breaches and auto-escalate.
  1807	
  1808	        This is the entry point for the SLA enforcement mechanism.
  1809	        Call periodically (e.g., every 5 minutes) or on-demand.
  1810	        """
  1811	        with _session_factory() as session:
  1812	            # Find all non-terminal findings
  1813	            terminal_states = {"closed", "false_positive"}
  1814	            findings = (
  1815	                session.query(AuditFinding)
  1816	                .filter(AuditFinding.state.isnot(None))
  1817	                .all()
  1818	            )
  1819	
  1820	            finding_dicts = [
  1821	                {
  1822	                    "id": f.id,
  1823	                    "state": f.state.value if f.state else "detected",
  1824	                    "severity": f.severity.value if f.severity else "medium",
  1825	                    "detected_at": f.state_last_changed_at or f.detected_at,
  1826	                    "finding_ref": f.finding_ref,
  1827	                }
  1828	                for f in findings
  1829	                if f.state and f.state.value not in terminal_states
  1830	            ]
  1831	
  1832	            escalations = _state_machine.check_timeouts(finding_dicts, session)
  1833	
  1834	            # Update findings that were escalated
  1835	            for esc in escalations:
  1836	                if esc.success:
  1837	                    finding = session.query(AuditFinding).filter(
  1838	                        AuditFinding.id == esc.finding_id
  1839	                    ).first()
  1840	                    if finding:
  1841	                        finding.state = FindingState.ESCALATED
  1842	                        finding.state_last_changed_at = datetime.now(timezone.utc)
  1843	
  1844	                        # Emit timeout breach event
  1845	                        _event_bus.publish(
  1846	                            EventType.FINDING_TIMEOUT_BREACH,
  1847	                            payload={
  1848	                                "finding_id": esc.finding_id,
  1849	                                "from_state": esc.from_state,
  1850	                                "to_state": "escalated",
  1851	                                "severity": esc.context.get("severity", "medium"),
  1852	                                "elapsed_hours": esc.context.get("elapsed_hours"),
  1853	                                "timeout_hours": esc.context.get("timeout_hours"),
  1854	                            },
  1855	                            source="state_machine",
  1856	                            correlation_id=esc.finding_id,
  1857	                        )
  1858	
  1859	            session.commit()
  1860	            _event_bus.process_pending()
  1861	
  1862	            return TimeoutCheckResponse(
  1863	                findings_checked=len(finding_dicts),
  1864	                auto_escalated=len(escalations),
  1865	                escalations=[
  1866	                    TransitionResponse(
