   202→   202→export default function ComplianceDashboard() {
   203→   203→  const [findings, setFindings] = useState<Finding[]>([]);
   204→   204→  const [mttr, setMttr] = useState<MTTRData | null>(null);

   209→   209→
   210→   210→  // Pipeline / trace state
   211→   211→  const [pingData, setPingData] = useState<PingData | null>(null);
   212→   212→  const [pingLoading, setPingLoading] = useState(false);
   213→   213→  const [pingError, setPingError] = useState<string | null>(null);

   215→   215→  const [postResult, setPostResult] = useState<string | null>(null);
   216→   216→  const [responseHeaders, setResponseHeaders] = useState<Record<string, string>>({});
   217→   217→  const [traceHistory, setTraceHistory] = useState<StoredTrace[]>([]);
   218→   218→  const [loadingHistory, setLoadingHistory] = useState(false);
   219→   219→  const renderStartRef = useRef<number>(0);
   220→   220→
   221→   221→  // ── Frontend Trace state (new) ──
   222→   222→  const {
   223→   223→    observedResources, stats: tracerStats,
   224→   224→    traceAndPersist, getResourceTiming,

   229→   229→  const [traceSummary, setTraceSummary] = useState<Record<string, unknown> | null>(null);
   230→   230→  const [traceSummaryLoading, setTraceSummaryLoading] = useState(false);
   231→   231→  const [fullTraceHistory, setFullTraceHistory] = useState<StoredTrace[]>([]);
   232→   232→  const [fullHistoryLoading, setFullHistoryLoading] = useState(false);
   233→   233→
   234→   234→  const fetchData = useCallback(async () => {
   235→   235→    setLoading(true);
   236→   236→    try {
   237→   237→      const [fRes, mRes, pRes, prRes] = await Promise.all([
   238→   238→        fetch('/api/compliance/findings'), fetch('/api/compliance/mttr'),

   250→   250→  useEffect(() => { fetchData(); }, [fetchData]);
   251→   251→

   253→   253→    setLoadingHistory(true);
   254→   254→    try {
   255→   255→      const res = await fetch('/api/system/ping?trace=history');
   256→   256→      const data = await res.json();

   258→   258→    } catch (e) { console.error(e); }
   259→   259→    finally { setLoadingHistory(false); }
   260→   260→  }, []);
   261→   261→
   262→   262→  const runAudit = async () => {
   263→   263→    setAuditing(true);
   264→   264→    try {
   265→   265→      const res = await fetch('/api/compliance/audit', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({}) });

   272→   272→  // ── Instrumented ping with browser-side Performance API ──
   273→   273→  const runPing = useCallback(async () => {
   274→   274→    setPingLoading(true);
   275→   275→    setPingError(null);
   276→   276→    setPostResult(null);
   277→   277→    renderStartRef.current = 0;
   278→   278→
   279→   279→    const fetchStart = performance.now();
   280→   280→    const fetchStartIso = new Date().toISOString();
   281→   281→

   283→   283→    const nav = navigator as Record<string, unknown>;
   284→   284→    const connection = (nav.connection as Record<string, unknown>) ?? {};

   289→   289→    try {
   290→   290→      const entries = performance.getEntriesByType?.('navigation') as Array<{ type: string }> | undefined;
   291→   291→      if (entries?.[0]) navigationType = entries[0].type;
   292→   292→    } catch { /* not available */ }
   293→   293→
   294→   294→    try {
   295→   295→      const res = await fetch('/api/system/ping');

   297→   297→
   298→   298→      const jsonStart = performance.now();

   305→   305→
   306→   306→      // Capture response headers

   309→   309→      setResponseHeaders(headers);

   311→   311→      // Now schedule state update timing measurement
   312→   312→      requestAnimationFrame(() => {
   313→   313→        const afterRender = Math.round(performance.now() - fetchStart);
   314→   314→        renderStartRef.current = afterRender;
   315→   315→      });
   316→   316→
   317→   317→      // Attach client timing to data so the waterfall can use it immediately
   318→   318→      const clientTimingData = {
   319→   319→        fetchStartIso,
   320→   320→        ttfbMs,
   321→   321→        roundTripMs,
   322→   322→        jsonParseMs,
   323→   323→        renderStartMs,
   324→   324→        networkProtocol: '',
   325→   325→        navigationType,
   326→   326→        connectionType,
   327→   327→      };

   329→   329→      // Merge into pingData for immediate waterfall display

   342→   342→        },

   344→   344→      }).then(r => r.json()).then(d => {
   345→   345→        setPostResult(JSON.stringify(d, null, 2));
   346→   346→      }).catch(() => {});
   347→   347→
   348→   348→      // Refresh trace history
   349→   349→      setTimeout(() => loadTraceHistory(), 300);
   350→   350→
   351→   351→    } catch (e) {
   352→   352→      setPingError(e instanceof Error ? e.message : 'Request failed');
   353→   353→    } finally {
   354→   354→      setPingLoading(false);
   355→   355→    }
   356→   356→  }, [loadTraceHistory]);
   357→   357→

   359→   359→  const runPostPing = useCallback(async () => {

   361→   361→    setPingError(null);
   362→   362→    setPostResult(null);
   363→   363→    const fetchStart = performance.now();

   365→   365→    try {
   366→   366→      const payload = postPayload.trim() ? JSON.parse(postPayload) : { test: true, timestamp: new Date().toISOString() };
   367→   367→      const res = await fetch('/api/system/ping', {
   368→   368→        method: 'POST',
   369→   369→        headers: {
   370→   370→          'Content-Type': 'application/json',
   371→   371→          'x-client-timing': JSON.stringify({
   372→   372→            fetchStartIso,
   373→   373→            ttfbMs: 0,
   374→   374→            roundTripMs: Math.round(performance.now() - fetchStart),
   375→   375→            jsonParseMs: 0,
   376→   376→            renderStartMs: Math.round(performance.now() - fetchStart),
   377→   377→            networkProtocol: '', navigationType: 'fetch', connectionType: 'unknown',
   378→   378→          }),
   379→   379→        },
   380→   380→        body: JSON.stringify(payload),
   381→   381→      });
   382→   382→      const data = await res.json();
   383→   383→      setPostResult(JSON.stringify(data, null, 2));
   384→   384→      setTimeout(() => loadTraceHistory(), 300);
   385→   385→    } catch (e) {
   386→   386→      setPingError(e instanceof Error ? e.message : 'POST failed');
   387→   387→    } finally {
   388→   388→      setPingLoading(false);
   389→   389→    }
   390→   390→  }, [postPayload, loadTraceHistory]);
   391→   391→
   392→   392→  // ── CORRELATED TRACE: trace+persist using the useRequestTracer hook ──
   393→   393→  const runCorrelatedTrace = useCallback(async () => {
   394→   394→    setCorrelatedLoading(true);
   395→   395→    setCorrelatedError(null);
   396→   396→    try {

   398→   398→        '/api/system/correlated-trace',
   399→   399→        {

   401→   401→          headers: { 'Content-Type': 'application/json' },
   402→   402→          body: JSON.stringify({ source: 'frontend_trace_tab', component: 'CorrelatedTracePanel' }),
   403→   403→          source: 'frontend_trace_tab',
   404→   404→          component: 'CorrelatedTracePanel',
